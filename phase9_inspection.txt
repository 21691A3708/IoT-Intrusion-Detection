from pathlib import Path

import joblib
import numpy as np
import pandas as pd

MODEL_FILE = Path("models/random_forest_capture_based.joblib")
TEST_FILE = Path("data/processed/splits/test.csv")
OUTPUT_DIR = Path("results/incident_response")

OUTPUT_DIR.mkdir(parents=True, exist_ok=True)

# Initial demonstration thresholds.
# These are configurable policy thresholds, not validated security standards.
LOW_MAX = 24
MEDIUM_MAX = 49
HIGH_MAX = 74

def assign_risk(score):
    if score <= LOW_MAX:
        return "LOW"
    if score <= MEDIUM_MAX:
        return "MEDIUM"
    if score <= HIGH_MAX:
        return "HIGH"
    return "CRITICAL"

def recommend_response(risk, predicted_label):
    if predicted_label == 0:
        return "LOG_ONLY"

    recommendations = {
        "LOW": "MONITOR_AND_LOG",
        "MEDIUM": "RAISE_ALERT_AND_MONITOR",
        "HIGH": "PRIORITIZE_ANALYST_REVIEW",
        "CRITICAL": "URGENT_ANALYST_REVIEW_AND_CONTAINMENT_ASSESSMENT",
    }
    return recommendations[risk]

print("Loading capture-separated model...")
saved = joblib.load(MODEL_FILE)
model = saved["model"]
feature_names = saved["feature_names"]

print("Loading sample records...")
test_df = pd.read_csv(TEST_FILE)

# The original test split has the features expected by the earlier model,
# but Number was excluded from the capture-separated model.
X = test_df.drop(columns=["Label"], errors="ignore")
X = X.reindex(columns=feature_names)

# Use a bounded sample for this initial demonstration.
sample = X.sample(
    n=min(1000, len(X)),
    random_state=42,
).copy()

if not np.isfinite(sample.to_numpy(dtype=float)).all():
    raise ValueError(
        "Non-finite values found. Clean the sample before prediction."
    )

print(f"Records to analyze: {len(sample)}")
print("Predicting attack probabilities...")

probabilities = model.predict_proba(sample)[:, 1]
predictions = model.predict(sample)

incidents = []

for index, (_, row) in enumerate(sample.iterrows()):
    probability = float(probabilities[index])
    predicted_label = int(predictions[index])

    # Risk score for this prototype: model attack probability * 100.
    score = round(probability * 100, 2)
    risk = assign_risk(score)

    # Rank the most influential features using Random Forest importance
    # multiplied by this record's normalized feature magnitude.
    # This is a heuristic, not a SHAP explanation.
    feature_values = row.to_numpy(dtype=float)
    importances = model.feature_importances_
    weighted_values = np.abs(feature_values) * importances

    top_indices = np.argsort(weighted_values)[-3:][::-1]
    top_features = ", ".join(
        feature_names[i] for i in top_indices
    )

    label = "UDP_FLOOD" if predicted_label == 1 else "NORMAL"

    incidents.append({
        "RecordIndex": int(row.name),
        "PredictedClass": label,
        "AttackProbability": round(probability, 6),
        "RiskScore": score,
        "RiskLevel": risk,
        "RecommendedAction": recommend_response(
            risk, predicted_label
        ),
        "TopHeuristicFeatures": top_features,
        "ResponseMode": "SIMULATION_ONLY",
    })

incident_df = pd.DataFrame(incidents)

output_file = OUTPUT_DIR / "incident_report.csv"
incident_df.to_csv(output_file, index=False)

print("\nRISK SUMMARY")
print("=" * 45)
print(incident_df["RiskLevel"].value_counts().to_string())

print("\nPREDICTION SUMMARY")
print(incident_df["PredictedClass"].value_counts().to_string())

print("\nSample incidents:")
print(incident_df.head(10).to_string(index=False))

print(f"\nIncident report saved to: {output_file}")
print("Simulation completed. No network actions were performed.")

from pathlib import Path

import joblib
import numpy as np
import pandas as pd
import shap

MODEL_FILE = Path("models/random_forest_capture_based.joblib")
TEST_FILE = Path("data/processed/splits/test.csv")
OUTPUT_DIR = Path("results/incident_response")

OUTPUT_DIR.mkdir(parents=True, exist_ok=True)

SAMPLE_SIZE = 20
BACKGROUND_SIZE = 100

def risk_level(score):
    if score < 25:
        return "LOW"
    if score < 50:
        return "MEDIUM"
    if score < 75:
        return "HIGH"
    return "CRITICAL"

def recommended_action(prediction, risk):
    if prediction == 0:
        return "LOG_ONLY"
    return {
        "LOW": "MONITOR_AND_LOG",
        "MEDIUM": "RAISE_ALERT_AND_MONITOR",
        "HIGH": "PRIORITIZE_ANALYST_REVIEW",
        "CRITICAL": "URGENT_ANALYST_REVIEW_AND_CONTAINMENT_ASSESSMENT",
    }[risk]

print("Loading capture-separated model...")
saved = joblib.load(MODEL_FILE)
model = saved["model"]
features = saved["feature_names"]

print("Loading test data...")
df = pd.read_csv(TEST_FILE)
X_all = df.reindex(columns=features)

# Select records predicted as attacks, up to SAMPLE_SIZE.
probabilities_all = model.predict_proba(X_all)[:, 1]
attack_indices = np.flatnonzero(probabilities_all >= 0.5)

if len(attack_indices) == 0:
    raise RuntimeError("No predicted attacks found in the test data.")

chosen = attack_indices[:SAMPLE_SIZE]
X_sample = X_all.iloc[chosen].copy()
probabilities = probabilities_all[chosen]
predictions = model.predict(X_sample)

background = X_all.sample(
    n=min(BACKGROUND_SIZE, len(X_all)),
    random_state=42,
)

print(f"Records to explain: {len(X_sample)}")
print("Creating SHAP explainer...")

explainer = shap.TreeExplainer(
    model,
    data=background,
    feature_perturbation="interventional",
    model_output="raw",
)

print("Calculating SHAP values...")
values = explainer.shap_values(
    X_sample,
    check_additivity=False,
)

# Extract SHAP contributions for class 1 (UDP Flood).
if isinstance(values, list):
    attack_values = values[1]
elif getattr(values, "ndim", 0) == 3:
    if values.shape[2] == 2:
        attack_values = values[:, :, 1]
    elif values.shape[1] == 2:
        attack_values = values[:, 1, :]
    else:
        raise ValueError(f"Unexpected SHAP shape: {values.shape}")
else:
    attack_values = values

if attack_values.shape != X_sample.shape:
    raise ValueError(
        f"Unexpected SHAP shape {attack_values.shape}; "
        f"expected {X_sample.shape}"
    )

incidents = []

for position, source_index in enumerate(chosen):
    probability = float(probabilities[position])
    prediction = int(predictions[position])
    score = round(probability * 100, 2)
    risk = risk_level(score)

    contributions = attack_values[position]

    top_positive_indices = np.argsort(contributions)[-3:][::-1]
    top_negative_indices = np.argsort(contributions)[:3]

    positive_text = "; ".join(
        f"{features[i]} ({contributions[i]:+.5f})"
        for i in top_positive_indices
        if contributions[i] > 0
    )

    negative_text = "; ".join(
        f"{features[i]} ({contributions[i]:+.5f})"
        for i in top_negative_indices
        if contributions[i] < 0
    )

    incidents.append({
        "RecordIndex": int(source_index),
        "PredictedClass": (
            "UDP_FLOOD" if prediction == 1 else "NORMAL"
        ),
        "AttackProbability": round(probability, 6),
        "RiskScore": score,
        "RiskLevel": risk,
        "RecommendedAction": recommended_action(prediction, risk),
        "TopPositiveSHAP": positive_text,
        "TopNegativeSHAP": negative_text,
        "ResponseMode": "LAB_ONLY_NOT_EXECUTED",
    })

report = pd.DataFrame(incidents)
output_file = OUTPUT_DIR / "integrated_incident_report.csv"
report.to_csv(output_file, index=False)

print("\nIntegrated incident report:")
print(report.head(10).to_string(index=False))
print(f"\nSaved report: {output_file}")
print("No containment actions were executed.")

class PredictionConnectionManager:

    def __init__(self):
        self.active_connections: list[WebSocket] = []

    async def connect(self, websocket: WebSocket):
        await websocket.accept()
        self.active_connections.append(websocket)
        print(
            f"WebSocket connected. Active clients: "
            f"{len(self.active_connections)}"
        )

    def disconnect(self, websocket: WebSocket):
        if websocket in self.active_connections:
            self.active_connections.remove(websocket)

        print(
            f"WebSocket disconnected. Active clients: "
            f"{len(self.active_connections)}"
        )

    async def broadcast(self, message: dict):
        disconnected = []

        for connection in self.active_connections.copy():
            try:
                await connection.send_json(message)
            except Exception:
                disconnected.append(connection)

        for connection in disconnected:
            self.disconnect(connection)

@app.post("/api/predictions")
async def create_prediction(request: PredictionRequest):
    # Step 1: Run the ML prediction
    try:
        result = predict_traffic(request.features)
    except ValueError as exc:
        raise HTTPException(
            status_code=422,
            detail=str(exc),
        ) from exc

    # Step 2: Save prediction and optionally create an incident
    try:
        with SessionLocal() as db:
            row = db.execute(
                text("""
                    INSERT INTO public.predictions (
                        device_id,
                        predicted_label,
                        attack_probability,
                        risk_level,
                        source,
                        features,
                        model_version
                    )
                    VALUES (
                        :device_id,
                        :predicted_label,
                        :attack_probability,
                        :risk_level,
                        :source,
                        CAST(:features AS jsonb),
                        :model_version
                    )
                    RETURNING
                        id,
                        device_id,
                        predicted_label,
                        attack_probability,
                        risk_level,
                        source,
                        model_version,
                        created_at
                """),
                {
                    "device_id": (
                        str(request.device_id)
                        if request.device_id
                        else None
                    ),
                    "predicted_label": result["predicted_label"],
                    "attack_probability": result["attack_probability"],
                    "risk_level": result["risk_level"],
                    "source": request.source,
                    "features": json.dumps(request.features),
                    "model_version": "random_forest_capture_based",
                },
            ).mappings().one()

            # Step 3: Automatically create an incident
            # when the risk level is HIGH or CRITICAL
            incident_id = None

            if result["risk_level"] in ("high", "critical"):
                incident = db.execute(
                    text("""
                        INSERT INTO public.incidents (
                            prediction_id,
                            device_id,
                            title,
                            description,
                            severity,
                            status
                        )
                        VALUES (
                            :prediction_id,
                            :device_id,
                            :title,
                            :description,
                            :severity,
                            'open'
                        )
                        RETURNING id
                    """),
                    {
                        "prediction_id": str(row["id"]),
                        "device_id": (
                            str(request.device_id)
                            if request.device_id
                            else None
                        ),
                        "title": (
                            f"{result['risk_level'].upper()} "
                            "IoT Security Alert"
                        ),
                        "description": (
                            "The intrusion detection model detected "
                            "an attack with "
                            f"{result['attack_probability']:.2%} "
                            "attack probability."
                        ),
                        "severity": result["risk_level"],
                    },
                ).mappings().one()

                incident_id = str(incident["id"])

            # Step 4: Commit prediction and incident together
            db.commit()
            response_data = {
                 **dict(row),
                 "prediction": result["prediction"],
                 "incident_created": incident_id is not None,
                 "incident_id": incident_id,
                 }

           # Step 5: Prepare prediction response
           # Step 6: Broadcast the new prediction to connected clients
            await prediction_manager.broadcast({
                "event": "new_prediction",
                "data": jsonable_encoder(response_data),
                })
            return response_data

    except IntegrityError as exc:
        raise HTTPException(
            status_code=400,
            detail=(
                "The supplied device ID is invalid, or a database "
                "constraint was violated."
            ),
        ) from exc


@app.get("/api/predictions")
def get_predictions(
    limit: int = Query(default=50, ge=1, le=1000),
    offset: int = Query(default=0, ge=0),
    risk_level: str | None = None,
    predicted_label: int | None = Query(default=None, ge=0, le=1),
):
    try:
        with SessionLocal() as db:

            # Query total number of matching predictions
            count_query = text("""
                SELECT COUNT(*)
                FROM public.predictions
                WHERE
                    (
                        CAST(:risk_level AS VARCHAR) IS NULL
                        OR risk_level = CAST(:risk_level AS VARCHAR)
                    )
                    AND
                    (
                        CAST(:predicted_label AS SMALLINT) IS NULL
                        OR predicted_label = CAST(:predicted_label AS SMALLINT)
                    )
            """)

            total = db.execute(
                count_query,
                {
                    "risk_level": risk_level,
                    "predicted_label": predicted_label,
                },
            ).scalar_one()

            # Retrieve prediction history
            history_query = text("""
                SELECT
                    id,
                    device_id,
                    predicted_label,
                    attack_probability,
                    risk_level,
                    source,
                    model_version,
                    created_at
                FROM public.predictions
                WHERE
                    (
                        CAST(:risk_level AS VARCHAR) IS NULL
                        OR risk_level = CAST(:risk_level AS VARCHAR)
                    )
                    AND
                    (
                        CAST(:predicted_label AS SMALLINT) IS NULL
                        OR predicted_label = CAST(:predicted_label AS SMALLINT)
                    )
                ORDER BY created_at DESC
                LIMIT :limit OFFSET :offset
            """)

            rows = db.execute(
                history_query,
                {
                    "risk_level": risk_level,
                    "predicted_label": predicted_label,
                    "limit": limit,
                    "offset": offset,
                },
            ).mappings().all()

            return {
                "total": total,
                "limit": limit,
                "offset": offset,
                "items": [dict(row) for row in rows],
            }

    except Exception as e:
        raise HTTPException(
            status_code=500,
            detail=f"Could not retrieve prediction history: {str(e)}",
        )

@app.patch("/api/incidents/{incident_id}/status")
def update_incident_status(
    incident_id: UUID,
    request: IncidentStatusUpdate
):
    try:
        with SessionLocal() as db:

            # 1. Find and lock the incident
            existing = db.execute(
                text("""
                    SELECT id, status
                    FROM public.incidents
                    WHERE id = :incident_id
                    FOR UPDATE
                """),
                {"incident_id": str(incident_id)}
            ).mappings().first()

            if existing is None:
                raise HTTPException(
                    status_code=404,
                    detail="Incident not found."
                )

            old_status = existing["status"]
            new_status = request.status

            # 2. Update the incident status
            updated = db.execute(
                text("""
                    UPDATE public.incidents
                    SET
                        status = :new_status,
                        resolved_at = CASE
                            WHEN :new_status = 'resolved'
                            THEN CURRENT_TIMESTAMP
                            ELSE NULL
                        END
                    WHERE id = :incident_id
                    RETURNING
                        id,
                        prediction_id,
                        device_id,
                        title,
                        severity,
                        status,
                        resolved_at,
                        created_at
                """),
                {
                    "incident_id": str(incident_id),
                    "new_status": new_status
                }
            ).mappings().one()

            # 3. Insert audit log
            audit = db.execute(
                text("""
                    INSERT INTO public.incident_audit_logs (
                        incident_id,
                        action,
                        details
                    )
                    VALUES (
                        :incident_id,
                        :action,
                        CAST(:details AS jsonb)
                    )
                    RETURNING id, created_at
                """),
                {
                    "incident_id": str(incident_id),
                    "action": "status_changed",
                    "details": json.dumps({
                        "old_status": old_status,
                        "new_status": new_status
                    })
                }
            ).mappings().one()

            # 4. Commit both operations together
            db.commit()

            return {
                "message": "Incident status updated successfully.",
                "incident": dict(updated),
                "audit_log": {
                    "id": str(audit["id"]),
                    "action": "status_changed",
                    "old_status": old_status,
                    "new_status": new_status,
                    "created_at": audit["created_at"]
                }
            }

    except HTTPException:
        raise

    except Exception as exc:
        raise HTTPException(
            status_code=500,
            detail=f"Could not update incident: {str(exc)}"
        ) from exc


@app.post("/api/upload/csv")
def upload_csv(file: UploadFile = File(...)):
    """
    Upload an IoT traffic CSV and perform batch intrusion detection.
    Predictions and incidents are saved in a single database transaction.
    """

    if not file.filename or not file.filename.lower().endswith(".csv"):
        raise HTTPException(
            status_code=400,
            detail="Please upload a valid CSV file."
        )

    MAX_FILE_SIZE = 100 * 1024 * 1024  # 100 MB
    CHUNK_SIZE = 1000
    MAX_ROWS = 500000

    try:
        # Check file size without loading the entire file into memory.
        file.file.seek(0, 2)
        file_size = file.file.tell()
        file.file.seek(0)

        if file_size == 0:
            raise HTTPException(
                status_code=400,
                detail="The uploaded CSV file is empty."
            )

        if file_size > MAX_FILE_SIZE:
            raise HTTPException(
                status_code=413,
                detail="CSV file exceeds the 100 MB upload limit."
            )

        # Read only the CSV header first.
        header = pd.read_csv(file.file, nrows=0, encoding="utf-8-sig")
        file.file.seek(0)

        if header.columns.duplicated().any():
            raise HTTPException(
                status_code=400,
                detail="CSV contains duplicate column names."
            )

        uploaded_columns = list(header.columns)

        missing_features = [
            name for name in EXPECTED_FEATURES
            if name not in uploaded_columns
        ]

        if missing_features:
            raise HTTPException(
                status_code=422,
                detail={
                    "message": "CSV is missing required model features.",
                    "missing_features": missing_features
                }
            )

        # These columns are not used by the ML model.
        ignored_columns = {
            "Number",
            "Label",
            "Telnet",
            "SMTP"
        }

        unexpected_columns = [
            name for name in uploaded_columns
            if name not in EXPECTED_FEATURES
            and name not in ignored_columns
        ]

        if unexpected_columns:
            raise HTTPException(
                status_code=422,
                detail={
                    "message": "CSV contains unsupported columns.",
                    "unexpected_columns": unexpected_columns
                }
            )

        summary = {
            "total_records": 0,
            "normal_records": 0,
            "attack_records": 0,
            "risk_distribution": {
                "low": 0,
                "medium": 0,
                "high": 0,
                "critical": 0
            },
            "incidents_created": 0
        }

        with SessionLocal() as db:
            try:
                # Process the CSV in chunks to control memory usage.
                chunks = pd.read_csv(
                    file.file,
                    usecols=EXPECTED_FEATURES,
                    chunksize=CHUNK_SIZE,
                    encoding="utf-8-sig"
                )

                for chunk in chunks:

                    if len(chunk) == 0:
                        continue

                    if summary["total_records"] + len(chunk) > MAX_ROWS:
                        raise HTTPException(
                            status_code=413,
                            detail="CSV exceeds the 500,000 record limit."
                        )

                    # Convert every feature to numeric.
                    for column in EXPECTED_FEATURES:
                        chunk[column] = pd.to_numeric(
                            chunk[column],
                            errors="coerce"
                        )

                    # Reject missing, invalid and infinite values.
                    values = chunk[EXPECTED_FEATURES].to_numpy(
                        dtype=float
                    )

                    if not np.isfinite(values).all():
                        raise HTTPException(
                            status_code=422,
                            detail=(
                                "CSV contains missing, non-numeric, "
                                "or infinite feature values."
                            )
                        )

                    # Run predictions for the complete chunk.
                    predictions = predict_batch(
                        chunk[EXPECTED_FEATURES]
                    )

                    # Save predictions and incidents.
                    for (_, feature_row), result in zip(
                        chunk.iterrows(),
                        predictions
                    ):
                        features = {
                            name: float(feature_row[name])
                            for name in EXPECTED_FEATURES
                        }

                        prediction_row = db.execute(
                            text("""
                                INSERT INTO public.predictions (
                                    device_id,
                                    predicted_label,
                                    attack_probability,
                                    risk_level,
                                    source,
                                    features,
                                    model_version
                                )
                                VALUES (
                                    NULL,
                                    :predicted_label,
                                    :attack_probability,
                                    :risk_level,
                                    'csv_upload',
                                    CAST(:features AS jsonb),
                                    'random_forest_capture_based'
                                )
                                RETURNING id
                            """),
                            {
                                "predicted_label": result["predicted_label"],
                                "attack_probability": result["attack_probability"],
                                "risk_level": result["risk_level"],
                                "features": json.dumps(features)
                            }
                        ).mappings().one()

                        summary["total_records"] += 1

                        if result["predicted_label"] == 1:
                            summary["attack_records"] += 1
                        else:
                            summary["normal_records"] += 1

                        risk = result["risk_level"]
                        summary["risk_distribution"][risk] += 1

                        # Preserve the existing HIGH/CRITICAL incident rule.
                        if risk in ("high", "critical"):
                            db.execute(
                                text("""
                                    INSERT INTO public.incidents (
                                        prediction_id,
                                        device_id,
                                        title,
                                        description,
                                        severity,
                                        status
                                    )
                                    VALUES (
                                        :prediction_id,
                                        NULL,
                                        :title,
                                        :description,
                                        :severity,
                                        'open'
                                    )
                                """),
                                {
                                    "prediction_id": str(
                                        prediction_row["id"]
                                    ),
                                    "title": (
                                        f"{risk.upper()} "
                                        "IoT Security Alert"
                                    ),
                                    "description": (
                                        "CSV batch analysis detected "
                                        "traffic with "
                                        f"{result['attack_probability']:.2%} "
                                        "attack probability."
                                    ),
                                    "severity": risk
                                }
                            )

                            summary["incidents_created"] += 1

                if summary["total_records"] == 0:
                    raise HTTPException(
                        status_code=400,
                        detail="The CSV does not contain any data records."
                    )

                # Commit all uploaded records together.
                db.commit()

            except Exception:
                db.rollback()
                raise

        return {
            "message": "CSV analysis completed successfully.",
            "filename": file.filename,
            **summary
        }

    except HTTPException:
        raise

    except (pd.errors.ParserError, UnicodeDecodeError) as exc:
        raise HTTPException(
            status_code=400,
            detail="Could not parse the uploaded CSV file."
        ) from exc

    except Exception as exc:
        raise HTTPException(
            status_code=500,
            detail=f"CSV processing failed: {str(exc)}"
        ) from exc

    finally:
        file.file.close()

import math
from numbers import Real

from .device_context_service import (
    get_device_context,
)

from .attack_impact_service import (
    calculate_attack_impact,
)

from .multiclass_ml_service import (
    predict_traffic,
)


RISK_POLICY_VERSION = "context_risk_v1"

RISK_WEIGHTS = {

    "criticality": 0.40,

    "attack_impact": 0.40,

    "exposure": 0.20,
}


# --------------------------------------------------
# INPUT VALIDATION
# --------------------------------------------------

def validate_unit_value(
    value,
    name: str,
) -> float:

    if (
        isinstance(value, bool)
        or not isinstance(value, Real)
    ):

        raise ValueError(
            f"{name} must be numeric."
        )

    value = float(value)

    if (
        not math.isfinite(value)
        or not 0 <= value <= 1
    ):

        raise ValueError(
            f"{name} must be between 0 and 1."
        )

    return value


# --------------------------------------------------
# RISK SCORE
# --------------------------------------------------

def calculate_risk_score(

    attack_probability: float,

    criticality: float,

    attack_impact: float,

    exposure: float,

) -> float:

    probability = validate_unit_value(

        attack_probability,

        "attack_probability",
    )

    criticality = validate_unit_value(

        criticality,

        "criticality",
    )

    attack_impact = validate_unit_value(

        attack_impact,

        "attack_impact",
    )

    exposure = validate_unit_value(

        exposure,

        "exposure",
    )

    context_score = (

        RISK_WEIGHTS["criticality"]
        * criticality

        + RISK_WEIGHTS["attack_impact"]
        * attack_impact

        + RISK_WEIGHTS["exposure"]
        * exposure
    )

    risk_score = (

        100 * probability * context_score
    )

    return float(risk_score)


# --------------------------------------------------
# SEVERITY CLASSIFICATION
# --------------------------------------------------

def get_risk_level(
    risk_score: float,
) -> str:

    if risk_score < 25:

        return "low"

    if risk_score < 50:

        return "medium"

    if risk_score < 75:

        return "high"

    return "critical"


# --------------------------------------------------
# NON-DISRUPTIVE RECOMMENDATION
# --------------------------------------------------

def recommend_action(

    risk_level: str,

    detection_status: str,

    context_known: bool,

) -> str:

    if risk_level == "critical":

        return "priority_investigation"

    if risk_level == "high":

        return "alert_and_investigate"

    # High attack confidence must not be ignored
    # simply because the device has low criticality.

    if detection_status == "high_confidence_attack":

        return "alert_and_investigate"

    if detection_status == "elevated_suspicion":

        return "review_suspicious_activity"

    if risk_level == "medium":

        return "review_suspicious_activity"

    if detection_status == "suspicious":

        return "monitor_suspicious_activity"

    if not context_known:

        return "monitor_and_verify_device"

    return "record_and_monitor"


# --------------------------------------------------
# CONTEXT-AWARE RISK ASSESSMENT
# --------------------------------------------------

def assess_prediction_risk(

    prediction: dict,

    device_id: str | None = None,

) -> dict:

    context = get_device_context(
        device_id
    )

    impact_result = calculate_attack_impact(

        prediction["class_probabilities"]
    )

    probability = validate_unit_value(

        prediction["calibrated_attack_probability"],

        "attack_probability",
    )

    criticality = context["criticality"]

    exposure = context["exposure"]

    attack_impact = impact_result[
        "expected_impact"
    ]

    risk_score = calculate_risk_score(

        attack_probability=probability,

        criticality=criticality,

        attack_impact=attack_impact,

        exposure=exposure,
    )

    risk_level = get_risk_level(
        risk_score
    )

    # Explain each factor's contribution
    # to the final risk score.

    criticality_contribution = (

        100 * probability

        * RISK_WEIGHTS["criticality"]

        * criticality
    )

    impact_contribution = (

        100 * probability

        * RISK_WEIGHTS["attack_impact"]

        * attack_impact
    )

    exposure_contribution = (

        100 * probability

        * RISK_WEIGHTS["exposure"]

        * exposure
    )

    contribution_sum = (

        criticality_contribution

        + impact_contribution

        + exposure_contribution
    )

    recommendation = recommend_action(

        risk_level=risk_level,

        detection_status=prediction[
            "detection_status"
        ],

        context_known=context[
            "context_known"
        ],
    )

    warnings = [

        "Risk weights and thresholds are "
        "provisional policy parameters.",

        "Device context is demonstration-only.",

        "Attack probability calibration "
        "has not been independently evaluated.",
    ]

    if not context["context_known"]:

        warnings.append(

            "Unknown device: fallback criticality "
            "and exposure were used."
        )

    if not impact_result["impact_available"]:

        warnings.append(

            "No meaningful raw attack-class "
            "probability mass was available "
            "for impact estimation."
        )

    if prediction["candidate_attack_family"] == "BruteForce":

        warnings.append(

            "BruteForce detection has no "
            "independent validation coverage."
        )

    return {

        "risk_policy_version":
            RISK_POLICY_VERSION,

        "risk_score":
            risk_score,

        "risk_level":
            risk_level,

        "attack_probability":
            probability,

        "device_context":
            context,

        "attack_impact":
            impact_result,

        "risk_factors": {

            "criticality":
                criticality,

            "attack_impact":
                attack_impact,

            "exposure":
                exposure,
        },

        "risk_weights":
            RISK_WEIGHTS.copy(),

        "risk_contributions": {

            "criticality":
                criticality_contribution,

            "attack_impact":
                impact_contribution,

            "exposure":
                exposure_contribution,
        },

        "risk_reconstruction": {

            "contribution_sum":
                contribution_sum,

            "error":
                abs(contribution_sum - risk_score),

            "passed":
                math.isclose(

                    contribution_sum,

                    risk_score,

                    abs_tol=1e-8,

                    rel_tol=0,
                ),
        },

        "recommended_action":
            recommendation,

        "requires_device_verification":
            not context["identity_verified"],

        "containment_authorized":
            False,

        "response_executed":
            False,

        "warnings":
            warnings,
    }

        "response_executed": False,
    }

import math
from numbers import Real


ATTACK_IMPACT = {

    "DDoS": 0.90,

    "DoS": 0.75,

    "Reconnaissance": 0.40,

    "BruteForce": 0.70,

    "Spoofing": 0.80,

    "Mirai": 0.95,

    "WebBased": 0.85,
}


def calculate_attack_impact(
    class_probabilities: dict,
) -> dict:

    if not isinstance(
        class_probabilities, dict
    ):

        raise ValueError(
            "Class probabilities must be a dictionary."
        )

    expected_classes = (
        set(ATTACK_IMPACT) | {"Benign"}
    )

    if set(class_probabilities) != expected_classes:

        raise ValueError(
            "Unexpected class probability schema."
        )

    probabilities = {}

    for family, value in class_probabilities.items():

        if (
            isinstance(value, bool)
            or not isinstance(value, Real)
        ):

            raise ValueError(
                f"Invalid probability for {family}."
            )

        probability = float(value)

        if (
            not math.isfinite(probability)
            or not 0 <= probability <= 1
        ):

            raise ValueError(
                f"Invalid probability for {family}."
            )

        probabilities[family] = probability

    if not math.isclose(
        sum(probabilities.values()),
        1.0,
        abs_tol=1e-6,
        rel_tol=0,
    ):

        raise ValueError(
            "Class probabilities must sum to one."
        )

    raw_attack_probability = sum(

        probabilities[family]

        for family in ATTACK_IMPACT
    )

    contributions = []

    if raw_attack_probability <= 1e-12:

        return {

            "expected_impact": 0.0,

            "raw_attack_probability":
                raw_attack_probability,

            "impact_contributions": [],

            "impact_available": False,

            "impact_policy_version":
                "attack_impact_v1",
        }

    expected_impact = 0.0

    for family, impact in ATTACK_IMPACT.items():

        conditional_probability = (

            probabilities[family]

            / raw_attack_probability
        )

        contribution = (

            conditional_probability * impact
        )

        expected_impact += contribution

        contributions.append({

            "attack_family": family,

            "class_probability":
                probabilities[family],

            "conditional_probability":
                conditional_probability,

            "impact_score": impact,

            "weighted_contribution":
                contribution,
        })

    contributions.sort(

        key=lambda item:
            item["weighted_contribution"],

        reverse=True,
    )

    return {

        "expected_impact":
            float(expected_impact),

        "raw_attack_probability":
            raw_attack_probability,

        "impact_contributions":
            contributions,

        "impact_available": True,

        "impact_policy_version":
            "attack_impact_v1",

        "class_probabilities_calibrated":
            False,
    }

@router.post("/assess-risk")
def multiclass_risk_assessment(

    request: RiskAssessmentRequest,

):

    try:

        result = assess_traffic_risk(

            features=request.features,

            device_id=request.device_id,
        )

        return {

            **result,

            "source": request.source,
        }

    except (ValueError, OverflowError) as error:

        raise HTTPException(

            status_code=422,

            detail=str(error),

        ) from error

    except Exception as error:

        logger.exception(

            "Context-aware risk assessment failed."
        )

        raise HTTPException(

            status_code=500,

            detail="Risk assessment failed.",

        ) from error
