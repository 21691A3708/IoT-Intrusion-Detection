import { Routes, UrlMatcher } from '@angular/router';
import { Dashboard } from './components/dashboard/dashboard';

const pages = new Set(['dashboard', 'detection', 'traffic-monitor', 'incidents', 'response-center', 'devices', 'explainability', 'reports', 'assistant']);
const pageMatcher: UrlMatcher = segments => segments.length === 1 && pages.has(segments[0].path)
  ? { consumed: segments, posParams: { page: segments[0] } } : null;

export const routes: Routes = [
  { path: '', redirectTo: 'dashboard', pathMatch: 'full' },
  { path: 'alerts', redirectTo: 'incidents', pathMatch: 'full' },
  { path: 'traffic', redirectTo: 'traffic-monitor', pathMatch: 'full' },
  { path: 'analysis', redirectTo: 'explainability', pathMatch: 'full' },
  { matcher: pageMatcher, component: Dashboard },
  { path: '**', component: Dashboard },
];

import { Component, OnInit, inject, signal, DestroyRef } from '@angular/core';

import { ActivatedRoute, RouterLink, RouterLinkActive } from '@angular/router';
import { Title } from '@angular/platform-browser';
import { takeUntilDestroyed } from '@angular/core/rxjs-interop';

import { Api, Device, DashboardAnalytics } from '../../services/api';

import { PredictionWebSocket } from '../../services/prediction-websocket';

import { BackendStatus } from '../backend-status/backend-status';
import { Devices } from '../devices/devices';
import { DeviceForm } from '../device-form/device-form';
import { Detection } from '../detection/detection';
import { PredictionHistory } from '../prediction-history/prediction-history';
import { Explainability } from '../explainability/explainability';
import { IncidentManagement } from '../incident-management/incident-management';
import { CsvUpload } from '../csv-upload/csv-upload';
import { PredictionTrends } from '../prediction-trends/prediction-trends';
import { AssistantChat } from '../assistant-chat/assistant-chat';

export type DashboardView = 'dashboard' | 'alerts' | 'devices' | 'traffic' | 'analysis' | 'reports' | 'detection' | 'response-center' | 'assistant' | 'not-found';

@Component({
  selector: 'app-dashboard',
  standalone: true,
  imports: [
    RouterLink,
    RouterLinkActive,
    BackendStatus,
    Devices,
    DeviceForm,
    Detection,
    PredictionHistory,
    Explainability,
    IncidentManagement,
    CsvUpload,
    PredictionTrends,
    AssistantChat,
  ],
  templateUrl: './dashboard.html',
  styleUrl: './dashboard.scss',
})
export class Dashboard implements OnInit {
  private readonly api = inject(Api);
  private readonly route = inject(ActivatedRoute);
  private readonly browserTitle = inject(Title);
  protected readonly menuOpen = signal(false);

  private readonly predictionWebSocket = inject(PredictionWebSocket);

  private readonly destroyRef = inject(DestroyRef);

  // Application title
  protected readonly title = signal('Network Intrusion Detection System');

  // Current sidebar page
  protected readonly activeView = signal<DashboardView>('dashboard');

  // Shared WebSocket connection status
  protected readonly websocketStatus = this.predictionWebSocket.status;

  // Device data
  protected readonly devices = signal<Device[]>([]);
  protected readonly devicesLoading = signal(false);
  protected readonly devicesError = signal('');

  // Dashboard analytics
  protected readonly analytics = signal<DashboardAnalytics | null>(null);
  protected readonly analyticsLoading = signal(false);
  protected readonly analyticsError = signal('');

  // --------------------------------------------------
  // INITIALIZATION
  // --------------------------------------------------

  ngOnInit(): void {
    this.route.paramMap.pipe(takeUntilDestroyed(this.destroyRef)).subscribe(params => {
      const views: Record<string, DashboardView> = {
        dashboard: 'dashboard', incidents: 'alerts', devices: 'devices',
        'traffic-monitor': 'traffic', explainability: 'analysis', reports: 'reports',
        detection: 'detection', 'response-center': 'response-center', assistant: 'assistant',
      };
      const page = params.get('page') ?? '';
      this.activeView.set(Object.prototype.hasOwnProperty.call(views, page) ? views[page] : 'not-found');
      this.menuOpen.set(false);
      this.browserTitle.setTitle(`${this.getPageTitle()} | IoT Sentinel`);
    });
    this.loadDevices();
    this.loadAnalytics();
    this.connectPredictionWebSocket();
  }

  // --------------------------------------------------
  // SIDEBAR NAVIGATION
  // --------------------------------------------------

  protected getPageTitle(): string {
    const titles: Record<DashboardView, string> = {
      dashboard: 'Security Overview',
      alerts: 'Security Alerts',
      devices: 'IoT Device Management',
      traffic: 'Traffic Monitoring',
      analysis: 'Explainable AI Analysis',
      reports: 'Reports and Batch Analysis',
      detection: 'Intrusion Detection',
      'response-center': 'Response Center',
      assistant: 'AI Assistant',
      'not-found': 'Page Not Found',
    };

    return titles[this.activeView()];
  }

  // --------------------------------------------------
  // LOAD REGISTERED DEVICES
  // --------------------------------------------------

  protected loadDevices(showLoading = true): void {
    if (showLoading) {
      this.devicesLoading.set(true);
    }

    this.devicesError.set('');

    this.api
      .getDevices()
      .pipe(takeUntilDestroyed(this.destroyRef))
      .subscribe({
        next: (response) => {
          this.devices.set(response);
          this.devicesLoading.set(false);
        },

        error: (error) => {
          console.error('Device loading failed:', error);

          this.devicesError.set('Could not load devices. Check the backend and retry.');

          this.devicesLoading.set(false);
        },
      });
  }

  // --------------------------------------------------
  // LOAD DASHBOARD ANALYTICS
  // --------------------------------------------------

  protected loadAnalytics(showLoading = true): void {
    if (showLoading) {
      this.analyticsLoading.set(true);
    }

    this.analyticsError.set('');

    this.api
      .getDashboardAnalytics()
      .pipe(takeUntilDestroyed(this.destroyRef))
      .subscribe({
        next: (response) => {
          this.analytics.set(response);
          this.analyticsLoading.set(false);
        },

        error: (error) => {
          console.error('Dashboard analytics loading failed:', error);

          this.analyticsError.set(
            'Could not load dashboard analytics. Check the backend and retry.',
          );

          this.analyticsLoading.set(false);
        },
      });
  }

  // --------------------------------------------------
  // REAL-TIME WEBSOCKET CONNECTION
  // --------------------------------------------------

  private connectPredictionWebSocket(): void {
    this.predictionWebSocket
      .connect()
      .pipe(takeUntilDestroyed(this.destroyRef))
      .subscribe({
        next: (event) => {
          if (event.event === 'new_prediction') {
            console.log('Real-time prediction received:', event.data);

            // Refresh analytics when a new prediction arrives.
            this.loadAnalytics(false);
          }
        },

        error: (error) => {
          console.error('Prediction WebSocket error:', error);
        },

        complete: () => {
          console.log('Prediction WebSocket stream completed.');
        },
      });
  }

  // --------------------------------------------------
  // MANUAL REFRESH
  // --------------------------------------------------

  protected refreshDashboard(): void {
    this.loadDevices();
    this.loadAnalytics();
  }
}

import {
  Component,
  OnInit,
  DestroyRef,
  inject,
  signal,
} from '@angular/core';

import { takeUntilDestroyed } from '@angular/core/rxjs-interop';
import { timeout } from 'rxjs';

import { Api, HealthResponse } from '../../services/api';
import { environment } from '../../../environments/environment';

@Component({
  selector: 'app-backend-status',
  standalone: true,
  templateUrl: './backend-status.html',
  styleUrl: './backend-status.scss',
})
export class BackendStatus implements OnInit {

  private readonly api = inject(Api);
  private readonly destroyRef = inject(DestroyRef);

  protected readonly apiStatus = signal('Disconnected');

  protected readonly health =
    signal<HealthResponse | null>(null);

  ngOnInit(): void {
    this.checkHealth();
  }

  protected checkHealth(): void {

    if (this.apiStatus() === 'Checking connection...') {
      return;
    }

    this.apiStatus.set('Checking connection...');
    this.health.set(null);

    this.api.getHealth().pipe(
      timeout(environment.requestTimeoutMs),
      takeUntilDestroyed(this.destroyRef)
    ).subscribe({

      next: (response) => {
        this.health.set(response);
        this.apiStatus.set('Connected');
      },

      error: () => {
        this.health.set(null);
        this.apiStatus.set('Disconnected');
      },

    });

  }

}

  }

  protected closeChat(): void {
    this.isOpen.set(false);
    this.errorMessage.set('');
  }

  protected sendQuestion(): void {
    const question = this.question.trim();

    if (!question || this.isSending()) {
      return;
    }

    this.messages.update((current) => [
      ...current,
      { role: 'user', text: question, timestamp: new Date() },
    ]);

    this.question = '';
    this.isSending.set(true);
    this.errorMessage.set('');

    this.http
      .post<ChatResponse>(this.apiUrl, { question })
      .pipe(takeUntilDestroyed(this.destroyRef))
      .subscribe({
        next: (response) => {
          this.messages.update((current) => [
            ...current,
            {
              role: 'assistant',
              text:
                response?.answer ||
                'The assistant returned an empty response. Please try again.',
              timestamp: new Date(),
            },
          ]);
          this.isSending.set(false);
        },
        error: (error) => {
          console.error('Assistant chat request failed:', error);

          this.errorMessage.set(
            'Could not connect to the assistant. Make sure the FastAPI backend is running.',
          );
          this.messages.update((current) => [
            ...current,
            {
              role: 'assistant',
              text:
                'I could not reach the backend. Please check that the FastAPI server is running and try again.',
              timestamp: new Date(),
            },
          ]);
          this.isSending.set(false);
        },
      });
  }

  protected onEnter(event: Event): void {
    const keyboardEvent = event as KeyboardEvent;

    if (!keyboardEvent.shiftKey) {
      keyboardEvent.preventDefault();
      this.sendQuestion();
    }
  }

  protected trackMessage(index: number, message: ChatMessage): string {
    return `${message.role}-${message.timestamp.getTime()}-${index}`;
  }
}
