import { Component, inject, output, signal } from '@angular/core';
import { FormsModule } from '@angular/forms';
import { Api, Device, DeviceCreate } from '../../services/api';

@Component({
  selector: 'app-device-form',
  standalone: true,
  imports: [FormsModule],
  templateUrl: './device-form.html',
  styleUrl: './device-form.scss',
})
export class DeviceForm {
  private readonly api = inject(Api);

  readonly deviceRegistered = output<Device>();

  protected readonly savingDevice = signal(false);
  protected readonly deviceMessage = signal('');
  protected readonly deviceMessageType = signal<'success' | 'error'>('success');

  protected deviceName = '';
  protected deviceIdentifier = '';
  protected deviceType = 'temperature_sensor';
  protected deviceLocation = '';
  protected deviceStatus: DeviceCreate['status'] = 'online';

  protected registerDevice(): void {
    this.deviceMessage.set('');

    const device: DeviceCreate = {
      device_name: this.deviceName.trim(),
      device_identifier: this.deviceIdentifier.trim(),
      device_type: this.deviceType.trim(),
      location: this.deviceLocation.trim(),
      status: this.deviceStatus,
    };

    if (!device.device_name || !device.device_identifier || !device.device_type) {
      this.showError('Please complete all required fields.');
      return;
    }

    this.savingDevice.set(true);

    this.api.createDevice(device).subscribe({
      next: (createdDevice) => {
        this.deviceMessageType.set('success');
        this.deviceMessage.set(`Device "${createdDevice.device_name}" registered successfully.`);

        this.resetForm();
        this.savingDevice.set(false);

        // Notify Dashboard that a device was registered.
        this.deviceRegistered.emit(createdDevice);
      },

      error: (error) => {
        if (error.status === 409) {
          this.showError('That device identifier already exists. Please use a unique identifier.');
        } else if (error.status === 422) {
          this.showError('The server rejected the device details. Check the required fields.');
        } else {
          this.showError('Could not register the device. Check the backend and try again.');
        }

        this.savingDevice.set(false);
      },
    });
  }

  private showError(message: string): void {
    this.deviceMessageType.set('error');
    this.deviceMessage.set(message);
  }

  private resetForm(): void {
    this.deviceName = '';
    this.deviceIdentifier = '';
    this.deviceType = 'temperature_sensor';
    this.deviceLocation = '';
    this.deviceStatus = 'online';
  }
}

    return this.http.get<HealthResponse>(`${this.baseUrl}/api/health`);
  }

  createDevice(device: DeviceCreate): Observable<Device> {
    return this.http.post<Device>(`${this.baseUrl}/api/devices`, device);
  }

  createPrediction(request: PredictionRequest): Observable<PredictionResponse> {
    return this.http.post<PredictionResponse>(`${this.baseUrl}/api/predictions`, request);
  }
  getPredictionHistory(
    limit: number = 50,
    offset: number = 0,
  ): Observable<PredictionHistoryResponse> {
    return this.http.get<PredictionHistoryResponse>(`${this.baseUrl}/api/predictions`, {
      params: {
        limit,
        offset,
      },
    });
  }
  explainTraffic(request: ExplainRequest): Observable<ExplainResponse> {
    return this.http.post<ExplainResponse>(`${this.baseUrl}/api/explain`, request);
  }

  getIncidents(): Observable<IncidentResponse> {
    return this.http.get<IncidentResponse>(`${this.baseUrl}/api/incidents`);
  }

  updateIncidentStatus(
    incidentId: string,
    status: Exclude<IncidentStatus, 'open'>,
  ): Observable<any> {
    return this.http.patch<any>(`${this.baseUrl}/api/incidents/${incidentId}/status`, { status });
  }
  uploadCsv(file: File) {
    const formData = new FormData();
    formData.append('file', file);

    return this.http.post<CsvUploadResponse>(`${this.baseUrl}/api/upload/csv`, formData);
  }

  getDashboardAnalytics(): Observable<DashboardAnalytics> {
    return new Observable((observer) => {
      let predictionsLoaded = false;
      let incidentsLoaded = false;

      let predictionData: PredictionHistoryResponse | null = null;
      let incidentData: IncidentResponse | null = null;

      const calculateAnalytics = () => {
        if (!predictionsLoaded || !incidentsLoaded) {
          return;
        }

        const predictions = predictionData?.items ?? [];
        const incidents = incidentData?.items ?? [];

        const analytics: DashboardAnalytics = {
          totalPredictions: predictions.length,

          normalPredictions: predictions.filter((item) => item.predicted_label === 0).length,

          attackPredictions: predictions.filter((item) => item.predicted_label === 1).length,

          activeIncidents: incidents.filter(
            (incident) => incident.status !== 'resolved' && incident.status !== 'false_positive',
          ).length,

          riskDistribution: {
            low: predictions.filter((item) => item.risk_level === 'low').length,

            medium: predictions.filter((item) => item.risk_level === 'medium').length,

            high: predictions.filter((item) => item.risk_level === 'high').length,

            critical: predictions.filter((item) => item.risk_level === 'critical').length,
          },
        };

        observer.next(analytics);
        observer.complete();
      };

      const predictionsSubscription = this.getPredictionHistory(1000, 0).subscribe({
        next: (response) => {
          predictionData = response;
          predictionsLoaded = true;
          calculateAnalytics();
        },
        error: (error) => observer.error(error),
      });

      const incidentsSubscription = this.getIncidents().subscribe({
        next: (response) => {
          incidentData = response;
          incidentsLoaded = true;
          calculateAnalytics();
        },
        error: (error) => observer.error(error),
      });

      return () => {
        predictionsSubscription.unsubscribe();
        incidentsSubscription.unsubscribe();
      };
    });
  }
}

import {
  Component,
  DestroyRef,
  OnInit,
  inject,
  signal,
  computed,
} from '@angular/core';

import { FormsModule } from '@angular/forms';
import { DecimalPipe } from '@angular/common';
import { takeUntilDestroyed } from '@angular/core/rxjs-interop';
import { finalize } from 'rxjs';

import {
  MulticlassApiService,
} from '../../core/services/multiclass-api.service';

import {
  NotificationService,
} from '../../core/services/notification.service';

import {
  ATTACK_LABELS,
} from '../../core/models/prediction.model';

import type {
  ModelInfo,
  TrafficFeatures,
  AttackFamily,
  MulticlassPredictionResponse,
} from '../../core/models/prediction.model';

import {
  getApiErrorMessage,
} from '../../shared/utilities/api-error';

@Component({
  selector: 'app-detection',
  standalone: true,
  imports: [FormsModule, DecimalPipe],
  templateUrl: './detection.html',
  styleUrl: './detection.scss',
})
export class Detection implements OnInit {

  private readonly api = inject(MulticlassApiService);

  private readonly notifications =
    inject(NotificationService);

  private readonly destroyRef = inject(DestroyRef);

  protected readonly modelInfo =
    signal<ModelInfo | null>(null);

  protected readonly modelLoading = signal(false);

  protected readonly modelError = signal('');

  protected readonly submitting = signal(false);

  protected readonly errorMessage = signal('');

  protected readonly result =
    signal<MulticlassPredictionResponse | null>(null);

  protected readonly families =
    Object.values(ATTACK_LABELS);

  protected trafficJson = '';

  protected readonly statusLabels: Record<string, string> = {
    low_probability: 'Low attack probability',
    suspicious: 'Suspicious traffic',
    elevated_suspicion: 'Elevated suspicion',
    high_confidence_attack: 'High-confidence attack',
  };

  protected readonly probabilityRows = computed(() => {

    const prediction = this.result();

    if (!prediction) {
      return [];
    }

    return this.families.map(family => ({
      family,
      probability: prediction.class_probabilities[family],
      predicted: prediction.predicted_family === family,
    }));

  });

  protected readonly unsupportedClass = computed(() => {

    const prediction = this.result();
    const model = this.modelInfo();

    if (!prediction || !model) {
      return false;
    }

    return !model.validation_supported_labels.includes(
      prediction.predicted_label
    );

  });

  ngOnInit(): void {
    this.loadModelInfo();
  }

  protected loadModelInfo(): void {

    if (this.modelLoading() || this.submitting()) {
      return;
    }

    this.modelLoading.set(true);
    this.modelError.set('');
    this.modelInfo.set(null);

    this.api.getModelInfo().pipe(
      takeUntilDestroyed(this.destroyRef),
      finalize(() => this.modelLoading.set(false))
    ).subscribe({

      next: info => {

        if (
          !Array.isArray(info.features) ||
          info.feature_count !== 36 ||
          info.features.length !== 36 ||
          info.class_count !== 8
        ) {
          this.modelError.set(
            'Unexpected model configuration.'
          );
          return;
        }

        this.modelInfo.set(info);

      },

      error: error => {
        this.modelError.set(getApiErrorMessage(error));
      },

    });

  }

  protected loadTemplate(): void {

    const model = this.modelInfo();

    if (!model || this.submitting()) {
      return;
    }

    const template = Object.fromEntries(
      model.features.map(feature => [feature, null])
    );

    this.trafficJson = JSON.stringify(
      template, null, 2
    );

    this.result.set(null);
    this.errorMessage.set('');

    this.notifications.info(
      'Replace every null value with a measured numerical feature.',
      'Feature template loaded'
    );

  }

  protected clearInput(): void {

    if (this.submitting()) {
      return;
    }

    this.trafficJson = '';
    this.errorMessage.set('');
    this.result.set(null);

  }

  protected onInputChange(value: string): void {

    this.trafficJson = value;
    this.errorMessage.set('');
    this.result.set(null);

  }

  private validateFeatures(): TrafficFeatures {

    const model = this.modelInfo();

    if (!model) {
      throw new Error(
        'Model information is unavailable.'
      );
    }

    let parsed: unknown;

    try {
      parsed = JSON.parse(this.trafficJson);
    } catch {
      throw new Error(
        'Invalid JSON. Enter a valid traffic feature object.'
      );
    }

    if (
      typeof parsed !== 'object' ||
      parsed === null ||
      Array.isArray(parsed)
    ) {
      throw new Error(
        'Traffic features must be a JSON object.'
      );
    }

    const input = parsed as Record<string, unknown>;

    const missing = model.features.filter(
      feature => !Object.prototype.hasOwnProperty.call(
        input, feature
      )
    );

    if (missing.length > 0) {
      throw new Error(
        `Missing features: ${missing.join(', ')}`
      );
    }

    const extra = Object.keys(input).filter(
      feature => !model.features.includes(feature)
    );

    if (extra.length > 0) {
      throw new Error(
        `Unexpected features: ${extra.join(', ')}`
      );
    }

    const invalid = model.features.filter(feature => {

      const value = input[feature];

      return (
        typeof value !== 'number' ||
        !Number.isFinite(value) ||
        Math.abs(value) > 3.4028234663852886e38
      );

    });

    if (invalid.length > 0) {
      throw new Error(
        `Invalid numerical values: ${invalid.join(', ')}`
      );
    }

    const ordered = Object.fromEntries(
      model.features.map(feature => [
        feature,
        input[feature],
      ])
    );

    return ordered as unknown as TrafficFeatures;

  }

  private validateResponse(
    response: MulticlassPredictionResponse
  ): boolean {

    const model = this.modelInfo();

    const validProbability = (value: unknown): boolean =>
      typeof value === 'number' &&
      Number.isFinite(value) &&
      value >= 0 &&
      value <= 1;

    if (
      !response ||
      !model ||
      response.model_version !== model.model_version ||
      ATTACK_LABELS[response.predicted_label] !==
        response.predicted_family ||
      !validProbability(response.prediction_confidence) ||
      !validProbability(response.raw_attack_probability) ||
      !validProbability(response.calibrated_attack_probability) ||
      !response.class_probabilities ||
      !this.families.every(family =>
        validProbability(
          response.class_probabilities[family]
        )
      ) ||
      !Object.prototype.hasOwnProperty.call(
        this.statusLabels,
        response.detection_status
      ) ||
      response.mode !== 'prediction_only' ||
      response.incident_created !== false ||
      response.response_executed !== false
    ) {
      return false;
    }

    const sum = this.families.reduce(
      (total, family) =>
        total + response.class_probabilities[family],
      0
    );

    return Math.abs(sum - 1) < 0.0001;

  }

  protected detectTraffic(): void {

    if (this.submitting() || !this.modelInfo()) {
      return;
    }

    this.errorMessage.set('');
    this.result.set(null);

    let features: TrafficFeatures;

    try {
      features = this.validateFeatures();
    } catch (error) {

      this.errorMessage.set(
        error instanceof Error
          ? error.message
          : 'Invalid traffic features.'
      );

      return;

    }

    this.submitting.set(true);

    this.api.predict({
      features,
      source: 'manual',
    }).pipe(

      takeUntilDestroyed(this.destroyRef),

      finalize(() => this.submitting.set(false))

    ).subscribe({

      next: response => {

        if (!this.validateResponse(response)) {

          this.errorMessage.set(
            'Unexpected prediction response. Check the backend model and API contract.'
          );

          return;

        }

        this.result.set(response);

        if (response.high_confidence_alert) {

          this.notifications.warning(
            `Predicted family: ${response.predicted_family}. Further assessment is recommended.`,
            'High-confidence attack'
          );

        } else {

          this.notifications.success(
            'Multiclass prediction completed.',
            'Analysis completed'
          );

        }

      },

      error: error => {

        this.errorMessage.set(
          getApiErrorMessage(error)
        );

      },

    });

  }

}
