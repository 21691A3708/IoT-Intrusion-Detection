# Load the saved model bundle.

if not MODEL_PATH.is_file():
    raise FileNotFoundError(
        f"Multiclass model not found: {MODEL_PATH}"
    )

BUNDLE = joblib.load(MODEL_PATH)

MODEL = BUNDLE["model"]

FEATURE_NAMES = list(
    BUNDLE["feature_names"]
)

LABELS = {
    int(key): value
    for key, value in BUNDLE["labels"].items()
}

MODEL_VERSION = BUNDLE["model_version"]

CALIBRATION = BUNDLE["calibration"]

MODEL.set_params(verbose=0)


# Validate model metadata.

if FEATURE_NAMES != EXPECTED_FEATURES:
    raise RuntimeError(
        "Multiclass feature schema mismatch."
    )

if LABELS != EXPECTED_LABELS:
    raise RuntimeError(
        "Multiclass label mapping mismatch."
    )

if MODEL.n_features_in_ != len(FEATURE_NAMES):
    raise RuntimeError(
        "Model feature count mismatch."
    )

if list(MODEL.feature_names_in_) != FEATURE_NAMES:
    raise RuntimeError(
        "Model feature order mismatch."
    )

if set(MODEL.classes_) != set(LABELS):
    raise RuntimeError(
        "Model class mapping mismatch."
    )

CLASS_POSITIONS = {
    int(label): index
    for index, label in enumerate(MODEL.classes_)
}


def validate_features(features: dict) -> pd.DataFrame:

    if not isinstance(features, dict):
        raise ValueError(
            "Features must be provided as a dictionary."
        )

    missing = [
        name
        for name in FEATURE_NAMES
        if name not in features
    ]

    extra = [
        name
        for name in features
        if name not in FEATURE_NAMES
    ]

    if missing:
        raise ValueError(
            f"Missing features: {missing}"
        )

    if extra:
        raise ValueError(
            f"Unexpected features: {extra}"
        )

    values = []

    for name in FEATURE_NAMES:

        value = features[name]

        if isinstance(value, (bool, np.bool_)):
            raise ValueError(
                f"{name} cannot be a boolean."
            )

        if not isinstance(value, Real):
            raise ValueError(
                f"{name} must be numeric."
            )

        value = float(value)

        if not math.isfinite(value):
            raise ValueError(
                f"{name} must be finite."
            )

        if abs(value) > np.finfo(np.float32).max:
            raise ValueError(
                f"{name} exceeds the float32 range."
            )

        values.append(value)

    return pd.DataFrame(
        [values],
        columns=FEATURE_NAMES,
        dtype="float32",
    )


def get_detection_status(
    attack_probability: float,
) -> str:

    if attack_probability >= HIGH_CONFIDENCE_THRESHOLD:
        return "high_confidence_attack"

    if attack_probability >= 0.75:
        return "elevated_suspicion"

    if attack_probability >= 0.50:
        return "suspicious"

    return "low_probability"


def format_prediction(probabilities) -> dict:

    probabilities = np.asarray(
        probabilities,
        dtype=float,
    )

    predicted_index = int(
        np.argmax(probabilities)
    )

    predicted_label = int(
        MODEL.classes_[predicted_index]
    )

    predicted_family = LABELS[
        predicted_label
    ]

    benign_index = CLASS_POSITIONS[0]

    benign_probability = float(
        probabilities[benign_index]
    )

    raw_attack_probability = float(
        np.clip(
            1.0 - benign_probability,
            0.0,
            1.0,
        )
    )

    calibrated_probability = (
        calibrate_attack_probability(
            raw_attack_probability,
            CALIBRATION,
        )
    )

    class_probabilities = {
        LABELS[int(label)]: float(
            probabilities[index]
        )
        for index, label in enumerate(MODEL.classes_)
    }

    # Most probable attack family,
    # even if Benign is the overall top class.

    attack_labels = [
        label
        for label in LABELS
        if label != 0
    ]

    candidate_label = max(
        attack_labels,
        key=lambda label: probabilities[
            CLASS_POSITIONS[label]
        ],
    )

    candidate_probability = float(
        probabilities[
            CLASS_POSITIONS[candidate_label]
        ]
    )

    candidate_family = (
        LABELS[candidate_label]
        if candidate_probability > 0
        else None
    )

    return {

        "predicted_label": predicted_label,

        "predicted_family": predicted_family,

        # Original classifier decision.
        "prediction": (
            "normal"
            if predicted_label == 0
            else "attack"
        ),

        "prediction_confidence": float(
            probabilities[predicted_index]
        ),

        "candidate_attack_family":
            candidate_family,

        "candidate_attack_probability":
            candidate_probability,

        "benign_probability":
            benign_probability,

        "raw_attack_probability":
            raw_attack_probability,

        "attack_probability":
            calibrated_probability,

        "calibrated_attack_probability":
            calibrated_probability,

        "class_probabilities":
            class_probabilities,

        "class_probabilities_calibrated":
            False,

        "detection_status":
            get_detection_status(
                calibrated_probability
            ),

        "high_confidence_alert": (
            calibrated_probability
            >= HIGH_CONFIDENCE_THRESHOLD
        ),

        "alert_threshold":
            HIGH_CONFIDENCE_THRESHOLD,

        "model_version":
            MODEL_VERSION,

        "calibration_scope":
            "binary_attack_probability",

        "calibration_independently_validated":
            False,
    }


def predict_traffic(features: dict) -> dict:

    frame = validate_features(features)

    probabilities = MODEL.predict_proba(
        frame
    )[0]

    return format_prediction(
        probabilities
    )


def predict_batch(records: list) -> list:

    if not isinstance(records, list):
        raise ValueError(
            "Records must be provided as a list."
        )

    if not records:
        return []

    frames = [
        validate_features(record)
        for record in records
    ]

    frame = pd.concat(
        frames,
        ignore_index=True,
    )

    probabilities = MODEL.predict_proba(
        frame
    )

    return [
        format_prediction(row)
        for row in probabilities
    ]


def get_model_info() -> dict:

    return {

        "model_version":
            MODEL_VERSION,

        "algorithm":
            "RandomForestClassifier",

        "configuration":
            BUNDLE.get("configuration"),

        "feature_count":
            len(FEATURE_NAMES),

        "features":
            FEATURE_NAMES,

        "class_count":
            len(LABELS),

        "labels":
            LABELS,

        "calibrated":
            BUNDLE.get("calibrated", False),

        "calibration_scope":
            BUNDLE.get("calibration_scope"),

        "validation_supported_labels":
            BUNDLE.get(
                "validation_supported_labels",
                [],
            ),

        "high_confidence_threshold":
            HIGH_CONFIDENCE_THRESHOLD,
    }

import math
from numbers import Real

from .device_context_service import (
    get_device_context,
)

from .attack_impact_service import (
    calculate_attack_impact,
)

from .multiclass_ml_service import (
    predict_traffic,
)


RISK_POLICY_VERSION = "context_risk_v1"

RISK_WEIGHTS = {

    "criticality": 0.40,

    "attack_impact": 0.40,

    "exposure": 0.20,
}


# --------------------------------------------------
# INPUT VALIDATION
# --------------------------------------------------

def validate_unit_value(
    value,
    name: str,
) -> float:

    if (
        isinstance(value, bool)
        or not isinstance(value, Real)
    ):

        raise ValueError(
            f"{name} must be numeric."
        )

    value = float(value)

    if (
        not math.isfinite(value)
        or not 0 <= value <= 1
    ):

        raise ValueError(
            f"{name} must be between 0 and 1."
        )

    return value


# --------------------------------------------------
# RISK SCORE
# --------------------------------------------------

def calculate_risk_score(

    attack_probability: float,

    criticality: float,

    attack_impact: float,

    exposure: float,

) -> float:

    probability = validate_unit_value(

        attack_probability,

        "attack_probability",
    )

    criticality = validate_unit_value(

        criticality,

        "criticality",
    )

    attack_impact = validate_unit_value(

        attack_impact,

        "attack_impact",
    )

    exposure = validate_unit_value(

        exposure,

        "exposure",
    )

    context_score = (

        RISK_WEIGHTS["criticality"]
        * criticality

        + RISK_WEIGHTS["attack_impact"]
        * attack_impact

        + RISK_WEIGHTS["exposure"]
        * exposure
    )

    risk_score = (

        100 * probability * context_score
    )

    return float(risk_score)


# --------------------------------------------------
# SEVERITY CLASSIFICATION
# --------------------------------------------------

def get_risk_level(
    risk_score: float,
) -> str:

    if risk_score < 25:

        return "low"

    if risk_score < 50:

        return "medium"

    if risk_score < 75:

        return "high"

    return "critical"


# --------------------------------------------------
# NON-DISRUPTIVE RECOMMENDATION
# --------------------------------------------------

def recommend_action(

    risk_level: str,

    detection_status: str,

    context_known: bool,

) -> str:

    if risk_level == "critical":

        return "priority_investigation"

    if risk_level == "high":

        return "alert_and_investigate"

    # High attack confidence must not be ignored
    # simply because the device has low criticality.

    if detection_status == "high_confidence_attack":

        return "alert_and_investigate"

    if detection_status == "elevated_suspicion":

        return "review_suspicious_activity"

    if risk_level == "medium":

        return "review_suspicious_activity"

    if detection_status == "suspicious":

        return "monitor_suspicious_activity"

    if not context_known:

        return "monitor_and_verify_device"

    return "record_and_monitor"


# --------------------------------------------------
# CONTEXT-AWARE RISK ASSESSMENT
# --------------------------------------------------

def assess_prediction_risk(

    prediction: dict,

    device_id: str | None = None,

) -> dict:

    context = get_device_context(
        device_id
    )

    impact_result = calculate_attack_impact(

        prediction["class_probabilities"]
    )

    probability = validate_unit_value(

        prediction["calibrated_attack_probability"],

        "attack_probability",
    )

    criticality = context["criticality"]

    exposure = context["exposure"]

    attack_impact = impact_result[
        "expected_impact"
    ]

    risk_score = calculate_risk_score(

        attack_probability=probability,

        criticality=criticality,

        attack_impact=attack_impact,

        exposure=exposure,
    )

    risk_level = get_risk_level(
        risk_score
    )

    # Explain each factor's contribution
    # to the final risk score.

    criticality_contribution = (

        100 * probability

        * RISK_WEIGHTS["criticality"]

        * criticality
    )

    impact_contribution = (

        100 * probability

        * RISK_WEIGHTS["attack_impact"]

        * attack_impact
    )

    exposure_contribution = (

        100 * probability

        * RISK_WEIGHTS["exposure"]

        * exposure
    )

    contribution_sum = (

        criticality_contribution

        + impact_contribution

        + exposure_contribution
    )

    recommendation = recommend_action(

        risk_level=risk_level,

        detection_status=prediction[
            "detection_status"
        ],

        context_known=context[
            "context_known"
        ],
    )

    warnings = [

        "Risk weights and thresholds are "
        "provisional policy parameters.",

        "Device context is demonstration-only.",

        "Attack probability calibration "
        "has not been independently evaluated.",
    ]

    if not context["context_known"]:

        warnings.append(

            "Unknown device: fallback criticality "
            "and exposure were used."
        )

    if not impact_result["impact_available"]:

        warnings.append(

            "No meaningful raw attack-class "
            "probability mass was available "
            "for impact estimation."
        )

    if prediction["candidate_attack_family"] == "BruteForce":

        warnings.append(

            "BruteForce detection has no "
            "independent validation coverage."
        )

    return {

        "risk_policy_version":
            RISK_POLICY_VERSION,

        "risk_score":
            risk_score,

        "risk_level":
            risk_level,

        "attack_probability":
            probability,

        "device_context":
            context,

        "attack_impact":
            impact_result,

        "risk_factors": {

            "criticality":
                criticality,

            "attack_impact":
                attack_impact,

            "exposure":
                exposure,
        },

        "risk_weights":
            RISK_WEIGHTS.copy(),

        "risk_contributions": {

            "criticality":
                criticality_contribution,

            "attack_impact":
                impact_contribution,

            "exposure":
                exposure_contribution,
        },

        "risk_reconstruction": {

            "contribution_sum":
                contribution_sum,

            "error":
                abs(contribution_sum - risk_score),

            "passed":
                math.isclose(

                    contribution_sum,

                    risk_score,

                    abs_tol=1e-8,

                    rel_tol=0,
                ),
        },

        "recommended_action":
            recommendation,

        "requires_device_verification":
            not context["identity_verified"],

        "containment_authorized":
            False,

        "response_executed":
            False,

        "warnings":
            warnings,
    }


# --------------------------------------------------
# COMPLETE PREDICTION AND RISK PIPELINE
# --------------------------------------------------

def assess_traffic_risk(

    features: dict,

    device_id: str | None = None,

) -> dict:

    prediction = predict_traffic(
        features
    )

    risk = assess_prediction_risk(

        prediction=prediction,

        device_id=device_id,
    )

    return {

        "prediction": prediction,

        "risk_assessment": risk,

        "mode": "risk_assessment_only",

        "incident_created": False,

        "response_executed": False,
    }

def get_device_context(
    device_id: str | None = None,
) -> dict:

    if device_id is not None:

        if not isinstance(device_id, str):

            raise ValueError(
                "device_id must be a string."
            )

        if not device_id.strip():

            raise ValueError(
                "device_id cannot be empty."
            )

    if device_id in DEVICE_PROFILES:

        context = deepcopy(
            DEVICE_PROFILES[device_id]
        )

        context.update({

            "context_known": True,

            "context_source":
                "demonstration_inventory",

            "identity_verified": False,

            "simulation_only": True,
        })

        return context

    context = deepcopy(
        UNKNOWN_DEVICE_PROFILE
    )

    context.update({

        "requested_device_id": device_id,

        "context_known": False,

        "context_source":
            "unknown_device_fallback",

        "identity_verified": False,

        "simulation_only": True,
    })

    return context

import math
from numbers import Real


ATTACK_IMPACT = {

    "DDoS": 0.90,

    "DoS": 0.75,

    "Reconnaissance": 0.40,

    "BruteForce": 0.70,

    "Spoofing": 0.80,

    "Mirai": 0.95,

    "WebBased": 0.85,
}


def calculate_attack_impact(
    class_probabilities: dict,
) -> dict:

    if not isinstance(
        class_probabilities, dict
    ):

        raise ValueError(
            "Class probabilities must be a dictionary."
        )

    expected_classes = (
        set(ATTACK_IMPACT) | {"Benign"}
    )

    if set(class_probabilities) != expected_classes:

        raise ValueError(
            "Unexpected class probability schema."
        )

    probabilities = {}

    for family, value in class_probabilities.items():

        if (
            isinstance(value, bool)
            or not isinstance(value, Real)
        ):

            raise ValueError(
                f"Invalid probability for {family}."
            )

        probability = float(value)

        if (
            not math.isfinite(probability)
            or not 0 <= probability <= 1
        ):

            raise ValueError(
                f"Invalid probability for {family}."
            )

        probabilities[family] = probability

    if not math.isclose(
        sum(probabilities.values()),
        1.0,
        abs_tol=1e-6,
        rel_tol=0,
    ):

        raise ValueError(
            "Class probabilities must sum to one."
        )

    raw_attack_probability = sum(

        probabilities[family]

        for family in ATTACK_IMPACT
    )

    contributions = []

    if raw_attack_probability <= 1e-12:

        return {

            "expected_impact": 0.0,

            "raw_attack_probability":
                raw_attack_probability,

            "impact_contributions": [],

            "impact_available": False,

            "impact_policy_version":
                "attack_impact_v1",
        }

    expected_impact = 0.0

    for family, impact in ATTACK_IMPACT.items():

        conditional_probability = (

            probabilities[family]

            / raw_attack_probability
        )

        contribution = (

            conditional_probability * impact
        )

        expected_impact += contribution

        contributions.append({

            "attack_family": family,

            "class_probability":
                probabilities[family],

            "conditional_probability":
                conditional_probability,

            "impact_score": impact,

            "weighted_contribution":
                contribution,
        })

    contributions.sort(

        key=lambda item:
            item["weighted_contribution"],

        reverse=True,
    )

    return {

        "expected_impact":
            float(expected_impact),

        "raw_attack_probability":
            raw_attack_probability,

        "impact_contributions":
            contributions,

        "impact_available": True,

        "impact_policy_version":
            "attack_impact_v1",

        "class_probabilities_calibrated":
            False,
    }

import math


def calibrate_attack_probability(
    raw_probability: float,
    parameters: dict,
) -> float:

    probability = float(raw_probability)

    if not math.isfinite(probability):
        raise ValueError(
            "Attack probability must be finite."
        )

    if not 0.0 <= probability <= 1.0:
        raise ValueError(
            "Attack probability must be between 0 and 1."
        )

    if parameters.get("method") != "sigmoid_on_logit":
        raise ValueError(
            "Unsupported calibration method."
        )

    coefficient = float(
        parameters["coefficient"]
    )

    intercept = float(
        parameters["intercept"]
    )

    epsilon = float(
        parameters.get("epsilon", 1e-6)
    )

    if not all(
        math.isfinite(value)
        for value in [
            coefficient,
            intercept,
            epsilon,
        ]
    ):
        raise ValueError(
            "Invalid calibration parameters."
        )

    if not 0 < epsilon < 0.5:
        raise ValueError(
            "Invalid calibration epsilon."
        )

    probability = min(
        max(probability, epsilon),
        1.0 - epsilon,
    )

    logit = math.log(
        probability / (1.0 - probability)
    )

    z = coefficient * logit + intercept

    # Numerically stable sigmoid.
    if z >= 0:
        return 1.0 / (
            1.0 + math.exp(-z)
        )

    exp_z = math.exp(z)

    return exp_z / (1.0 + exp_z)

class PredictionConnectionManager:

    def __init__(self):
        self.active_connections: list[WebSocket] = []

    async def connect(self, websocket: WebSocket):
        await websocket.accept()
        self.active_connections.append(websocket)
        print(
            f"WebSocket connected. Active clients: "
            f"{len(self.active_connections)}"
        )

    def disconnect(self, websocket: WebSocket):
        if websocket in self.active_connections:
            self.active_connections.remove(websocket)

        print(
            f"WebSocket disconnected. Active clients: "
            f"{len(self.active_connections)}"
        )

    async def broadcast(self, message: dict):
        disconnected = []

        for connection in self.active_connections.copy():
            try:
                await connection.send_json(message)
            except Exception:
                disconnected.append(connection)

        for connection in disconnected:
            self.disconnect(connection)

@app.patch("/api/incidents/{incident_id}/status")
def update_incident_status(
    incident_id: UUID,
    request: IncidentStatusUpdate
):
    try:
        with SessionLocal() as db:

            # 1. Find and lock the incident
            existing = db.execute(
                text("""
                    SELECT id, status
                    FROM public.incidents
                    WHERE id = :incident_id
                    FOR UPDATE
                """),
                {"incident_id": str(incident_id)}
            ).mappings().first()

            if existing is None:
                raise HTTPException(
                    status_code=404,
                    detail="Incident not found."
                )

            old_status = existing["status"]
            new_status = request.status

            # 2. Update the incident status
            updated = db.execute(
                text("""
                    UPDATE public.incidents
                    SET
                        status = :new_status,
                        resolved_at = CASE
                            WHEN :new_status = 'resolved'
                            THEN CURRENT_TIMESTAMP
                            ELSE NULL
                        END
                    WHERE id = :incident_id
                    RETURNING
                        id,
                        prediction_id,
                        device_id,
                        title,
                        severity,
                        status,
                        resolved_at,
                        created_at
                """),
                {
                    "incident_id": str(incident_id),
                    "new_status": new_status
                }
            ).mappings().one()

            # 3. Insert audit log
            audit = db.execute(
                text("""
                    INSERT INTO public.incident_audit_logs (
                        incident_id,
                        action,
                        details
                    )
                    VALUES (
                        :incident_id,
                        :action,
                        CAST(:details AS jsonb)
                    )
                    RETURNING id, created_at
                """),
                {
                    "incident_id": str(incident_id),
                    "action": "status_changed",
                    "details": json.dumps({
                        "old_status": old_status,
                        "new_status": new_status
                    })
                }
            ).mappings().one()

            # 4. Commit both operations together
            db.commit()

            return {
                "message": "Incident status updated successfully.",
                "incident": dict(updated),
                "audit_log": {
                    "id": str(audit["id"]),
                    "action": "status_changed",
                    "old_status": old_status,
                    "new_status": new_status,
                    "created_at": audit["created_at"]
                }
            }

    except HTTPException:
        raise

    except Exception as exc:
        raise HTTPException(
            status_code=500,
            detail=f"Could not update incident: {str(exc)}"
        ) from exc


@app.post("/api/upload/csv")
def upload_csv(file: UploadFile = File(...)):
    """
    Upload an IoT traffic CSV and perform batch intrusion detection.
    Predictions and incidents are saved in a single database transaction.
    """

    if not file.filename or not file.filename.lower().endswith(".csv"):
        raise HTTPException(
            status_code=400,
            detail="Please upload a valid CSV file."
        )

    MAX_FILE_SIZE = 100 * 1024 * 1024  # 100 MB
    CHUNK_SIZE = 1000
    MAX_ROWS = 500000

    try:
        # Check file size without loading the entire file into memory.
        file.file.seek(0, 2)
        file_size = file.file.tell()
        file.file.seek(0)

        if file_size == 0:
            raise HTTPException(
                status_code=400,
                detail="The uploaded CSV file is empty."
            )

        if file_size > MAX_FILE_SIZE:
            raise HTTPException(
                status_code=413,
                detail="CSV file exceeds the 100 MB upload limit."
            )

        # Read only the CSV header first.
        header = pd.read_csv(file.file, nrows=0, encoding="utf-8-sig")
        file.file.seek(0)

        if header.columns.duplicated().any():
            raise HTTPException(
                status_code=400,
                detail="CSV contains duplicate column names."
            )

        uploaded_columns = list(header.columns)

        missing_features = [
            name for name in EXPECTED_FEATURES
            if name not in uploaded_columns
        ]

        if missing_features:
            raise HTTPException(
                status_code=422,
                detail={
                    "message": "CSV is missing required model features.",
                    "missing_features": missing_features
                }
            )

        # These columns are not used by the ML model.
        ignored_columns = {
            "Number",
            "Label",
            "Telnet",
            "SMTP"
        }

        unexpected_columns = [
            name for name in uploaded_columns
            if name not in EXPECTED_FEATURES
            and name not in ignored_columns
        ]

        if unexpected_columns:
            raise HTTPException(
                status_code=422,
                detail={
                    "message": "CSV contains unsupported columns.",
                    "unexpected_columns": unexpected_columns
                }
            )

        summary = {
            "total_records": 0,
            "normal_records": 0,
            "attack_records": 0,
            "risk_distribution": {
                "low": 0,
                "medium": 0,
                "high": 0,
                "critical": 0
            },
            "incidents_created": 0
        }

        with SessionLocal() as db:
            try:
                # Process the CSV in chunks to control memory usage.
                chunks = pd.read_csv(
                    file.file,
                    usecols=EXPECTED_FEATURES,
                    chunksize=CHUNK_SIZE,
                    encoding="utf-8-sig"
                )

                for chunk in chunks:

                    if len(chunk) == 0:
                        continue

                    if summary["total_records"] + len(chunk) > MAX_ROWS:
                        raise HTTPException(
                            status_code=413,
                            detail="CSV exceeds the 500,000 record limit."
                        )

                    # Convert every feature to numeric.
                    for column in EXPECTED_FEATURES:
                        chunk[column] = pd.to_numeric(
                            chunk[column],
                            errors="coerce"
                        )

                    # Reject missing, invalid and infinite values.
                    values = chunk[EXPECTED_FEATURES].to_numpy(
                        dtype=float
                    )

                    if not np.isfinite(values).all():
                        raise HTTPException(
                            status_code=422,
                            detail=(
                                "CSV contains missing, non-numeric, "
                                "or infinite feature values."
                            )
                        )

                    # Run predictions for the complete chunk.
                    predictions = predict_batch(
                        chunk[EXPECTED_FEATURES]
                    )

                    # Save predictions and incidents.
                    for (_, feature_row), result in zip(
                        chunk.iterrows(),
                        predictions
                    ):
                        features = {
                            name: float(feature_row[name])
                            for name in EXPECTED_FEATURES
                        }

                        prediction_row = db.execute(
                            text("""
                                INSERT INTO public.predictions (

                                    source,

                                    :risk_level,
                                    'csv_upload',
                                    CAST(:features AS jsonb),
                                    'random_forest_capture_based'
                                )
                                RETURNING id
                            """),
                            {
                                "predicted_label": result["predicted_label"],
                                "attack_probability": result["attack_probability"],
                                "risk_level": result["risk_level"],
                                "features": json.dumps(features)
                            }
                        ).mappings().one()

                        summary["total_records"] += 1

                        if result["predicted_label"] == 1:
                            summary["attack_records"] += 1
                        else:
                            summary["normal_records"] += 1

                        risk = result["risk_level"]
                        summary["risk_distribution"][risk] += 1

                        # Preserve the existing HIGH/CRITICAL incident rule.
                        if risk in ("high", "critical"):
                            db.execute(

                if summary["total_records"] == 0:

import {
  Component,
  DestroyRef,
  OnInit,
  inject,
  signal,
  computed,
  effect,
} from '@angular/core';

import { FormsModule } from '@angular/forms';
import { DecimalPipe } from '@angular/common';

import {
  DeviceContextSelector,
} from '../../shared/components/device-context-selector/device-context-selector';
import { takeUntilDestroyed } from '@angular/core/rxjs-interop';
import { finalize, TimeoutError } from 'rxjs';
import { RiskAssessmentApiService } from '../../core/services/risk-assessment-api.service';
import { DeviceContextService } from '../../core/services/device-context.service';
import { RiskScore } from '../../shared/components/risk-score/risk-score';
import { RiskContributions } from '../../shared/components/risk-contributions/risk-contributions';
import type { RiskAssessmentResponse } from '../../core/models/risk-assessment.model';

import {
  MulticlassApiService,
} from '../../core/services/multiclass-api.service';

import {
  NotificationService,
} from '../../core/services/notification.service';

import {
  ATTACK_LABELS,
} from '../../core/models/prediction.model';

import type {
  ModelInfo,
  TrafficFeatures,

} from '../../core/models/prediction.model';

import {
  getApiErrorMessage,
} from '../../shared/utilities/api-error';

@Component({
  selector: 'app-detection',
  standalone: true,
  imports: [
    FormsModule,
    DecimalPipe,
    DeviceContextSelector,
    RiskScore,
    RiskContributions,
  ],
  templateUrl: './detection.html',
  styleUrl: './detection.scss',
})
export class Detection implements OnInit {

  private readonly api = inject(MulticlassApiService);
  private readonly riskApi = inject(RiskAssessmentApiService);
  protected readonly context = inject(DeviceContextService);

  private readonly notifications =
    inject(NotificationService);

  private readonly destroyRef = inject(DestroyRef);

  protected readonly modelInfo =
    signal<ModelInfo | null>(null);

  protected readonly modelLoading = signal(false);

  protected readonly modelError = signal('');

  protected readonly submitting = signal(false);

  protected readonly errorMessage = signal('');

  private readonly assessmentState = signal<RiskAssessmentResponse | null>(null);
  private readonly submittedContext = signal<string | null>(null);
  protected readonly assessment = computed(() =>
    this.submittedContext() === this.context.selectedDeviceId()
      ? this.assessmentState() : null
  );
  protected readonly result = computed(() => this.assessment()?.prediction ?? null);

  constructor() {
    effect(() => {
      const selected = this.context.selectedDeviceId();
      if (selected !== this.submittedContext()) {
        this.assessmentState.set(null);
      }
    });
  }

  protected readonly families =
    Object.values(ATTACK_LABELS);

  protected trafficJson = '';

  protected readonly statusLabels: Record<string, string> = {
    low_probability: 'Low attack probability',
    suspicious: 'Suspicious traffic',
    elevated_suspicion: 'Elevated suspicion',
    high_confidence_attack: 'High-confidence attack',
  };

  protected readonly probabilityRows = computed(() => {

    const prediction = this.result();

    if (!prediction) {
      return [];
    }

    return this.families.map(family => ({
      family,
      probability: prediction.class_probabilities[family],
      predicted: prediction.predicted_family === family,
    }));

  });

  protected readonly unsupportedClass = computed(() => {

    const prediction = this.result();
    const model = this.modelInfo();

    if (!prediction || !model) {
      return false;
    }

    return !model.validation_supported_labels.includes(
      prediction.predicted_label
    );

  });

  ngOnInit(): void {
    this.loadModelInfo();
  }

  protected loadModelInfo(): void {

    if (this.modelLoading() || this.submitting()) {
      return;
    }

    this.modelLoading.set(true);
    this.modelError.set('');
    this.modelInfo.set(null);
    this.assessmentState.set(null);

    this.api.getModelInfo().pipe(
      takeUntilDestroyed(this.destroyRef),
      finalize(() => this.modelLoading.set(false))
    ).subscribe({

      next: info => {

        if (
          !Array.isArray(info.features) ||
          info.feature_count !== 36 ||
          info.features.length !== 36 ||
          info.class_count !== 8
        ) {
          this.modelError.set(
            'Unexpected model configuration.'
          );
          return;
        }

        this.modelInfo.set(info);

      },

      error: error => {
        this.modelError.set(getApiErrorMessage(error));
      },

    });

  }

  protected loadTemplate(): void {

    const model = this.modelInfo();

    if (!model || this.submitting()) {
      return;
    }

    const template = Object.fromEntries(
      model.features.map(feature => [feature, null])
    );

    this.trafficJson = JSON.stringify(
      template, null, 2
    );

    this.assessmentState.set(null);
    this.errorMessage.set('');

    this.notifications.info(
      'Replace every null value with a measured numerical feature.',
      'Feature template loaded'
    );

  }

  protected clearInput(): void {

    if (this.submitting()) {
      return;
    }

    this.trafficJson = '';
    this.errorMessage.set('');
    this.assessmentState.set(null);

  }

  protected onInputChange(value: string): void {

    this.trafficJson = value;
    this.errorMessage.set('');
    this.assessmentState.set(null);

  }

  private validateFeatures(): TrafficFeatures {

    const model = this.modelInfo();

    if (!model) {
      throw new Error(
        'Model information is unavailable.'
      );
    }

    let parsed: unknown;

    try {
      parsed = JSON.parse(this.trafficJson);
    } catch {
      throw new Error(
        'Invalid JSON. Enter a valid traffic feature object.'
      );
    }

    if (
      typeof parsed !== 'object' ||
      parsed === null ||
      Array.isArray(parsed)
    ) {
      throw new Error(
        'Traffic features must be a JSON object.'
      );
    }

    const input = parsed as Record<string, unknown>;

    const missing = model.features.filter(
      feature => !Object.prototype.hasOwnProperty.call(
        input, feature
      )
    );

    if (missing.length > 0) {
      throw new Error(
        `Missing features: ${missing.join(', ')}`
      );
    }

    const extra = Object.keys(input).filter(
      feature => !model.features.includes(feature)
    );

    if (extra.length > 0) {
      throw new Error(
        `Unexpected features: ${extra.join(', ')}`
      );
    }

    const invalid = model.features.filter(feature => {

      const value = input[feature];

      return (
        typeof value !== 'number' ||
        !Number.isFinite(value) ||
        Math.abs(value) > 3.4028234663852886e38
      );

    });

    if (invalid.length > 0) {
      throw new Error(
        `Invalid numerical values: ${invalid.join(', ')}`
      );
    }

    const ordered = Object.fromEntries(
      model.features.map(feature => [
        feature,
        input[feature],
      ])
    );

    return ordered as unknown as TrafficFeatures;

  }

  protected detectTraffic(): void {

    if (this.submitting() || !this.modelInfo() || this.context.loading()) {
      return;
    }

    this.errorMessage.set('');
    this.assessmentState.set(null);

    let features: TrafficFeatures;

    try {
      features = this.validateFeatures();
    } catch (error) {

      this.errorMessage.set(
        error instanceof Error
          ? error.message
          : 'Invalid traffic features.'
      );

      return;

    }

    const deviceId = this.context.selectedDeviceId();
    this.submittedContext.set(deviceId);
    this.submitting.set(true);

    this.riskApi.assessRisk({
      features,
      device_id: deviceId,
      source: 'manual',
    }).pipe(

      takeUntilDestroyed(this.destroyRef),

      finalize(() => this.submitting.set(false))

    ).subscribe({

      next: response => {

        if (deviceId !== this.context.selectedDeviceId()) {
          this.errorMessage.set('Device context changed. Analyze again.');
          return;
        }

        if (response.prediction.model_version !== this.modelInfo()?.model_version) {

          this.errorMessage.set(
            'Unexpected prediction response. Check the backend model and API contract.'
          );

          return;

        }

        this.assessmentState.set(response);
        const prediction = response.prediction;

        if (prediction.high_confidence_alert) {

          this.notifications.warning(
            `Predicted family: ${prediction.predicted_family}. Further assessment is recommended.`,
            'High-confidence attack'
          );

        } else {

          this.notifications.success(
            'Multiclass prediction and risk assessment completed.',
            'Analysis completed'
          );

        }

      },

      error: error => {

        const message = error instanceof Error &&
          error.message.startsWith('Invalid risk assessment response:')
          ? error.message : getApiErrorMessage(error);
        this.errorMessage.set(message);
        if (error instanceof TimeoutError || message.startsWith('Invalid risk assessment response:')) {
          this.notifications.error(message, 'Assessment failed');
        }

      },

    });

  }

}

import { Injectable, inject } from '@angular/core';
import { HttpClient } from '@angular/common/http';

  Observable,
  map,
  timeout,
} from 'rxjs';

import { environment } from '../../../environments/environment';

import {
  RISK_ACTION_LABELS,
  type RiskAssessmentRequest,
  type RiskAssessmentResponse,
} from '../models/risk-assessment.model';


const EPSILON = 1e-6;

const RISK_KEYS = [
  'criticality',
  'attack_impact',
  'exposure',
] as const;


// --------------------------------------------------
// VALIDATION HELPERS
// --------------------------------------------------

function requireValid(
  condition: unknown,
  message: string,
): asserts condition {

  if (!condition) {

    throw new Error(
      `Invalid risk assessment response: ${message}`
    );

  }

}


function asObject(
  value: unknown,
  name: string,
): Record<string, unknown> {

  requireValid(
    typeof value === 'object' &&
    value !== null &&
    !Array.isArray(value),
    `${name} must be an object`,
  );

  return value as Record<string, unknown>;

}


function asNumber(
  value: unknown,
  name: string,
  max = 1,
): number {

  requireValid(
    typeof value === 'number' &&
    Number.isFinite(value) &&
    value >= 0 &&
    value <= max + EPSILON,
    `${name} is outside its valid range`,
  );

  return value;

}


function asText(
  value: unknown,
  name: string,
): string {

  requireValid(
    typeof value === 'string' &&
    value.trim().length > 0,
    `${name} must be a non-empty string`,
  );

  return value;

}


function near(
  actual: number,
  expected: number,
  name: string,
): void {

  requireValid(
    Math.abs(actual - expected) <= EPSILON,
    `${name} is inconsistent`,
  );

}


// --------------------------------------------------
// RISK RESPONSE VALIDATION
// --------------------------------------------------

export function validateRiskAssessmentResponse(
  value: unknown,
  request: RiskAssessmentRequest,
): RiskAssessmentResponse {

  const response = asObject(value, 'response');

  const prediction = asObject(
    response['prediction'],
    'prediction',
  );

  const risk = asObject(
    response['risk_assessment'],
    'risk_assessment',
  );

  const context = asObject(
    risk['device_context'],
    'device_context',
  );

  const impact = asObject(
    risk['attack_impact'],
    'attack_impact',
  );

  const factors = asObject(
    risk['risk_factors'],
    'risk_factors',
  );

  const weights = asObject(
    risk['risk_weights'],
    'risk_weights',
  );

  const contributions = asObject(
    risk['risk_contributions'],
    'risk_contributions',
  );

  const reconstruction = asObject(
    risk['risk_reconstruction'],
    'risk_reconstruction',
  );


  // ------------------------------------------------
  // RESPONSE MODE AND SAFETY
  // ------------------------------------------------

  requireValid(
    response['mode'] === 'risk_assessment_only',
    'unexpected response mode',
  );

  requireValid(
    response['incident_created'] === false &&
    response['response_executed'] === false &&
    risk['response_executed'] === false &&
    risk['containment_authorized'] === false,
    'assessment-only safety contract violated',
  );

  requireValid(
    context['identity_verified'] === false &&
    context['simulation_only'] === true &&
    risk['requires_device_verification'] === true,
    'device safety contract violated',
  );

  requireValid(
    response['source'] === (request.source ?? 'manual'),
    'response source mismatch',
  );


  // ------------------------------------------------
  // PREDICTION VALIDATION
  // ------------------------------------------------

  const families = Object.values(ATTACK_LABELS);

  const label = prediction['predicted_label'];

  requireValid(
    typeof label === 'number' &&
    Number.isInteger(label) &&
    label >= 0 &&
    label <= 7,
    'invalid predicted label',
  );

  const family = prediction['predicted_family'];

  requireValid(
    family === ATTACK_LABELS[
      label as keyof typeof ATTACK_LABELS
    ],
    'predicted family does not match label',
  );

  requireValid(
    prediction['prediction'] ===
      (label === 0 ? 'normal' : 'attack'),
    'prediction type does not match label',
  );

  const probabilities = asObject(
    prediction['class_probabilities'],
    'class_probabilities',
  );

  requireValid(
    Object.keys(probabilities).length === families.length,
    'unexpected class probability count',
  );

  let probabilitySum = 0;
  let rawAttackMass = 0;

  for (const attackFamily of families) {

    const probability = asNumber(
      probabilities[attackFamily],
      attackFamily,
    );

    probabilitySum += probability;

    if (attackFamily !== 'Benign') {
      rawAttackMass += probability;
    }

  }

  near(probabilitySum, 1, 'class probability sum');

  const confidence = asNumber(
    prediction['prediction_confidence'],
    'prediction confidence',
  );

  near(
    confidence,
    asNumber(probabilities[String(family)], 'family probability'),
    'prediction confidence',
  );

  const candidate = prediction['candidate_attack_family'];

  requireValid(
    families.some(f => f !== 'Benign' && f === candidate),
    'invalid candidate attack family',
  );

  near(
    asNumber(
      prediction['candidate_attack_probability'],
      'candidate attack probability',
    ),
    asNumber(probabilities[String(candidate)], 'candidate probability'),
    'candidate attack probability',
  );

  near(
    asNumber(prediction['benign_probability'], 'benign probability'),
    asNumber(probabilities['Benign'], 'Benign'),
    'benign probability',
  );

  near(
    asNumber(prediction['raw_attack_probability'], 'raw attack probability'),
    rawAttackMass,
    'raw attack probability',
  );

  const probability = asNumber(
    prediction['calibrated_attack_probability'],
    'calibrated attack probability',
  );

  near(
    asNumber(prediction['attack_probability'], 'attack probability'),
    probability,
    'prediction attack probability',
  );

  near(
    asNumber(risk['attack_probability'], 'risk attack probability'),
    probability,
    'risk attack probability',
  );

  requireValid(
    prediction['class_probabilities_calibrated'] === false &&
    prediction['calibration_independently_validated'] === false &&
    prediction['calibration_scope'] === 'binary_attack_probability',
    'unexpected calibration metadata',
  );

  asText(prediction['model_version'], 'model version');

  const expectedDetectionStatus =
    probability >= 0.90 ? 'high_confidence_attack' :
    probability >= 0.75 ? 'elevated_suspicion' :
    probability >= 0.50 ? 'suspicious' :
    'low_probability';

  requireValid(
    prediction['detection_status'] === expectedDetectionStatus &&
    prediction['high_confidence_alert'] === (probability >= 0.90),
    'inconsistent detection status',
  );

  near(
    asNumber(prediction['alert_threshold'], 'alert threshold'),
    0.90,
    'alert threshold',
  );


  // ------------------------------------------------
  // DEVICE CONTEXT VALIDATION
  // ------------------------------------------------

  const criticality = asNumber(
    context['criticality'],
    'device criticality',
  );

  const exposure = asNumber(
    context['exposure'],
    'device exposure',
  );

  asText(context['device_name'], 'device name');
  asText(context['device_type'], 'device type');
  asText(context['business_function'], 'business function');

  requireValid(
    typeof context['context_known'] === 'boolean' &&
    typeof context['containment_allowed'] === 'boolean',
    'invalid device context flags',
  );

  if (context['context_known'] === true) {

    requireValid(
      typeof context['device_id'] === 'string' &&
      context['device_id'] === request.device_id &&
      context['context_source'] === 'demonstration_inventory',
      'selected device context mismatch',
    );

  } else {

    requireValid(
      context['device_id'] === null &&
      context['requested_device_id'] === (request.device_id ?? null) &&
      context['context_source'] === 'unknown_device_fallback' &&
      context['containment_allowed'] === false,
      'invalid unknown-device fallback',
    );

    near(criticality, 0.50, 'fallback criticality');
    near(exposure, 0.50, 'fallback exposure');

  }


  // ------------------------------------------------
  // EXPECTED ATTACK IMPACT
  // ------------------------------------------------

  const expectedImpact = asNumber(
    impact['expected_impact'],
    'expected attack impact',
  );

  requireValid(
    impact['impact_policy_version'] === 'attack_impact_v1',
    'unsupported attack impact policy',
  );

  near(
    asNumber(impact['raw_attack_probability'], 'impact attack mass'),
    rawAttackMass,
    'impact attack mass',
  );

  const impactRows = impact['impact_contributions'];

  requireValid(
    Array.isArray(impactRows),
    'impact contributions must be an array',
  );

  if (rawAttackMass <= 1e-12) {

    requireValid(
      impact['impact_available'] === false &&
      impactRows.length === 0,
      'invalid unavailable-impact response',
    );

    near(expectedImpact, 0, 'unavailable attack impact');

  } else {

    requireValid(
      impact['impact_available'] === true &&
      impact['class_probabilities_calibrated'] === false &&
      impactRows.length === 7,
      'invalid attack impact metadata',
    );

    const seen = new Set<string>();

    let impactSum = 0;
    let conditionalSum = 0;

    for (const item of impactRows) {

      const row = asObject(item, 'impact contribution');

      const attackFamily = asText(
        row['attack_family'],
        'impact attack family',
      );

      requireValid(
        families.some(f => f !== 'Benign' && f === attackFamily) &&
        !seen.has(attackFamily),
        'invalid or duplicate impact family',
      );

      seen.add(attackFamily);

      const classProbability = asNumber(
        row['class_probability'],
        'impact class probability',
      );

      const conditional = asNumber(
        row['conditional_probability'],
        'conditional probability',
      );

      const impactScore = asNumber(
        row['impact_score'],
        'family impact score',
      );

      const weighted = asNumber(
        row['weighted_contribution'],
        'weighted impact contribution',
      );

      near(
        classProbability,
        asNumber(probabilities[attackFamily], attackFamily),
        'impact class probability',
      );

      near(
        conditional,
        classProbability / rawAttackMass,
        'conditional attack probability',
      );

      near(
        weighted,
        conditional * impactScore,
        'weighted attack impact',
      );

      conditionalSum += conditional;
      impactSum += weighted;

    }

    near(conditionalSum, 1, 'conditional probability sum');
    near(impactSum, expectedImpact, 'expected attack impact');

  }


  // ------------------------------------------------
  // RISK CALCULATION VALIDATION
  // ------------------------------------------------

  requireValid(
    risk['risk_policy_version'] === 'context_risk_v1',
    'unsupported risk policy',
  );

  const score = asNumber(
    risk['risk_score'],
    'risk score',
    100,
  );

  const expectedLevel =
    score >= 75 ? 'critical' :
    score >= 50 ? 'high' :
    score >= 25 ? 'medium' :
    'low';

  requireValid(
    risk['risk_level'] === expectedLevel,
    'risk level does not match score',
  );

  const expectedFactors = {
    criticality,
    attack_impact: expectedImpact,
    exposure,
  };

  const expectedWeights = {
    criticality: 0.40,
    attack_impact: 0.40,
    exposure: 0.20,
  };

  let reconstructedScore = 0;

  for (const key of RISK_KEYS) {

    const factor = asNumber(
      factors[key],
      `${key} factor`,
    );

    const weight = asNumber(
      weights[key],
      `${key} weight`,
    );

    const contribution = asNumber(
      contributions[key],
      `${key} contribution`,
      100,
    );

    near(factor, expectedFactors[key], `${key} factor`);

    near(weight, expectedWeights[key], `${key} weight`);

    near(
      contribution,
      100 * probability * weight * factor,
      `${key} contribution`,
    );

    reconstructedScore += contribution;

  }

  near(reconstructedScore, score, 'risk score reconstruction');

  near(
    asNumber(reconstruction['contribution_sum'], 'contribution sum', 100),
    reconstructedScore,
    'reported contribution sum',
  );

  const reconstructionError = asNumber(
    reconstruction['error'],
    'reconstruction error',
    100,
  );

  requireValid(
    reconstruction['passed'] === true &&
    reconstructionError <= 1e-8,
    'backend risk reconstruction failed',
  );


  // ------------------------------------------------
  // RECOMMENDATIONS AND WARNINGS
  // ------------------------------------------------

  const action = asText(
    risk['recommended_action'],
    'recommended action',
  );

  requireValid(
    Object.prototype.hasOwnProperty.call(
      RISK_ACTION_LABELS,
      action,
    ),
    'unsupported recommended action',
  );

  const warnings = risk['warnings'];

  requireValid(
    Array.isArray(warnings) &&
    warnings.every(
      warning =>
        typeof warning === 'string' &&
        warning.trim().length > 0,
    ),
    'invalid risk warnings',
  );

  return value as RiskAssessmentResponse;

}

@Injectable({
  providedIn: 'root',
})
export class RiskAssessmentApiService {

  private readonly http = inject(HttpClient);


  assessRisk(
    request: RiskAssessmentRequest,
  ): Observable<RiskAssessmentResponse> {

    return this.http.post<unknown>(

      API_ENDPOINTS.risk.assess,

      request,

    ).pipe(

      timeout(environment.analysisTimeoutMs),

      map(response =>
        validateRiskAssessmentResponse(response, request)
      ),

    );

  }

}

import {
  Injectable,
  inject,
  signal,
  computed,
} from '@angular/core';

import { finalize } from 'rxjs';

import { DeviceApiService } from './device-api.service';

import type {
  DeviceContextProfile,
} from '../models/device.model';

import {
  getApiErrorMessage,
} from '../../shared/utilities/api-error';

@Injectable({
  providedIn: 'root',
})
export class DeviceContextService {

  private readonly api = inject(DeviceApiService);

  private readonly profileState =
    signal<DeviceContextProfile[]>([]);

  private readonly selectedIdState =
    signal<string | null>(null);

  private readonly loadingState = signal(false);

  private readonly loadedState = signal(false);

  private readonly errorState = signal('');

  readonly profiles = this.profileState.asReadonly();

  readonly selectedDeviceId =
    this.selectedIdState.asReadonly();

  readonly loading = this.loadingState.asReadonly();

  readonly loaded = this.loadedState.asReadonly();

  readonly error = this.errorState.asReadonly();

  readonly selectedProfile = computed(() => {

    const selectedId = this.selectedIdState();

    return this.profileState().find(
      profile => profile.device_id === selectedId
    ) ?? null;

  });

  readonly totalProfiles = computed(
    () => this.profileState().length
  );

  readonly restrictedProfiles = computed(
    () => this.profileState().filter(
      profile => !profile.containment_allowed
    ).length
  );

  readonly verifiedProfiles = computed(
    () => this.profileState().filter(
      profile => profile.identity_verified
    ).length
  );

  loadProfiles(force = false): void {

    if (
      this.loadingState() ||
      (this.loadedState() && !force)
    ) {
      return;
    }

    this.loadingState.set(true);
    this.errorState.set('');

    this.api.getDeviceProfiles().pipe(

      finalize(() => this.loadingState.set(false))

    ).subscribe({

      next: response => {

        this.profileState.set(response.devices);
        this.loadedState.set(true);

        const selectedId = this.selectedIdState();

        if (
          selectedId !== null &&
          !response.devices.some(
            profile => profile.device_id === selectedId
          )
        ) {
          this.selectedIdState.set(null);
        }

      },

      error: error => {

        this.profileState.set([]);
        this.selectedIdState.set(null);
        this.loadedState.set(false);

        this.errorState.set(
          getApiErrorMessage(error)
        );

      },

    });

  }

  selectDevice(deviceId: string | null): boolean {

    if (deviceId === null) {

      this.selectedIdState.set(null);

      return true;

    }

    if (
      this.loadingState() ||
      !this.loadedState() ||
      !this.profileState().some(
        profile => profile.device_id === deviceId
      )
    ) {
      return false;
    }

    this.selectedIdState.set(deviceId);

    return true;

  }

  clearSelection(): void {

    this.selectedIdState.set(null);

  }

}
