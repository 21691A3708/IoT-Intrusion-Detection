RISK_POLICY_VERSION = "context_risk_v1"

RISK_WEIGHTS = {

    "criticality": 0.40,

    "attack_impact": 0.40,

    "exposure": 0.20,
}


# --------------------------------------------------
# INPUT VALIDATION
# --------------------------------------------------

def validate_unit_value(
    value,
    name: str,
) -> float:

    if (
        isinstance(value, bool)
        or not isinstance(value, Real)
    ):

        raise ValueError(
            f"{name} must be numeric."
        )

    value = float(value)

    if (
        not math.isfinite(value)
        or not 0 <= value <= 1
    ):

        raise ValueError(
            f"{name} must be between 0 and 1."
        )

    return value


# --------------------------------------------------
# RISK SCORE
# --------------------------------------------------

def calculate_risk_score(

    attack_probability: float,

    criticality: float,

    attack_impact: float,

    exposure: float,

) -> float:

    probability = validate_unit_value(

        attack_probability,

        "attack_probability",
    )

    criticality = validate_unit_value(

        criticality,

        "criticality",
    )

    attack_impact = validate_unit_value(

        attack_impact,

        "attack_impact",
    )

    exposure = validate_unit_value(

        exposure,

        "exposure",
    )

    context_score = (

        RISK_WEIGHTS["criticality"]
        * criticality

        + RISK_WEIGHTS["attack_impact"]
        * attack_impact

        + RISK_WEIGHTS["exposure"]
        * exposure
    )

    risk_score = (

        100 * probability * context_score
    )

    return float(risk_score)


# --------------------------------------------------
# SEVERITY CLASSIFICATION
# --------------------------------------------------

def get_risk_level(
    risk_score: float,
) -> str:

    if risk_score < 25:

        return "low"

    if risk_score < 50:

        return "medium"

    if risk_score < 75:

        return "high"

    return "critical"


# --------------------------------------------------
# NON-DISRUPTIVE RECOMMENDATION
# --------------------------------------------------

def recommend_action(

    risk_level: str,

    detection_status: str,

    context_known: bool,

) -> str:

    if risk_level == "critical":

        return "priority_investigation"

    if risk_level == "high":

        return "alert_and_investigate"

    # High attack confidence must not be ignored
    # simply because the device has low criticality.

    if detection_status == "high_confidence_attack":

        return "alert_and_investigate"

    if detection_status == "elevated_suspicion":

        return "review_suspicious_activity"

    if risk_level == "medium":

        return "review_suspicious_activity"

    if detection_status == "suspicious":

        return "monitor_suspicious_activity"

    if not context_known:

        return "monitor_and_verify_device"

    return "record_and_monitor"


# --------------------------------------------------
# CONTEXT-AWARE RISK ASSESSMENT
# --------------------------------------------------

def assess_prediction_risk(

    prediction: dict,

    device_id: str | None = None,

) -> dict:

    context = get_device_context(
        device_id
    )

    impact_result = calculate_attack_impact(

        prediction["class_probabilities"]
    )

    probability = validate_unit_value(

        prediction["calibrated_attack_probability"],

        "attack_probability",
    )

    criticality = context["criticality"]

    exposure = context["exposure"]

    attack_impact = impact_result[
        "expected_impact"
    ]

    risk_score = calculate_risk_score(

        attack_probability=probability,

        criticality=criticality,

        attack_impact=attack_impact,

        exposure=exposure,
    )

    risk_level = get_risk_level(
        risk_score
    )

    # Explain each factor's contribution
    # to the final risk score.

    criticality_contribution = (

        100 * probability

        * RISK_WEIGHTS["criticality"]

        * criticality
    )

    impact_contribution = (

        100 * probability

        * RISK_WEIGHTS["attack_impact"]

        * attack_impact
    )

    exposure_contribution = (

        100 * probability

        * RISK_WEIGHTS["exposure"]

        * exposure
    )

    contribution_sum = (

        criticality_contribution

        + impact_contribution

        + exposure_contribution
    )

    recommendation = recommend_action(

        risk_level=risk_level,

        detection_status=prediction[
            "detection_status"
        ],

        context_known=context[
            "context_known"
        ],
    )

    warnings = [

        "Risk weights and thresholds are "
        "provisional policy parameters.",

        "Device context is demonstration-only.",

        "Attack probability calibration "
        "has not been independently evaluated.",
    ]

    if not context["context_known"]:

        warnings.append(

            "Unknown device: fallback criticality "
            "and exposure were used."
        )

    if not impact_result["impact_available"]:

        warnings.append(

            "No meaningful raw attack-class "
            "probability mass was available "
            "for impact estimation."
        )

    if prediction["candidate_attack_family"] == "BruteForce":

        warnings.append(

            "BruteForce detection has no "
            "independent validation coverage."
        )

    return {

        "risk_policy_version":
            RISK_POLICY_VERSION,

        "risk_score":
            risk_score,

        "risk_level":
            risk_level,

        "attack_probability":
            probability,

        "device_context":
            context,

        "attack_impact":
            impact_result,

        "risk_factors": {

            "criticality":
                criticality,

            "attack_impact":
                attack_impact,

            "exposure":
                exposure,
        },

        "risk_weights":
            RISK_WEIGHTS.copy(),

        "risk_contributions": {

            "criticality":
                criticality_contribution,

            "attack_impact":
                impact_contribution,

            "exposure":
                exposure_contribution,
        },

        "risk_reconstruction": {

            "contribution_sum":
                contribution_sum,

            "error":
                abs(contribution_sum - risk_score),

            "passed":
                math.isclose(

                    contribution_sum,

                    risk_score,

                    abs_tol=1e-8,

                    rel_tol=0,
                ),
        },

        "recommended_action":
            recommendation,

        "requires_device_verification":
            not context["identity_verified"],

        "containment_authorized":
            False,

        "response_executed":
            False,

        "warnings":
            warnings,
    }


# --------------------------------------------------
# COMPLETE PREDICTION AND RISK PIPELINE
# --------------------------------------------------

def assess_traffic_risk(

    features: dict,

    device_id: str | None = None,

) -> dict:

    prediction = predict_traffic(
        features
    )

    risk = assess_prediction_risk(

        prediction=prediction,

        device_id=device_id,
    )

    return {

        "prediction": prediction,

        "risk_assessment": risk,

        "mode": "risk_assessment_only",

        "incident_created": False,

        "response_executed": False,
    }

import math
from numbers import Real


ATTACK_IMPACT = {

    "DDoS": 0.90,

    "DoS": 0.75,

    "Reconnaissance": 0.40,

    "BruteForce": 0.70,

    "Spoofing": 0.80,

    "Mirai": 0.95,

    "WebBased": 0.85,
}


def calculate_attack_impact(
    class_probabilities: dict,
) -> dict:

    if not isinstance(
        class_probabilities, dict
    ):

        raise ValueError(
            "Class probabilities must be a dictionary."
        )

    expected_classes = (
        set(ATTACK_IMPACT) | {"Benign"}
    )

    if set(class_probabilities) != expected_classes:

        raise ValueError(
            "Unexpected class probability schema."
        )

    probabilities = {}

    for family, value in class_probabilities.items():

        if (
            isinstance(value, bool)
            or not isinstance(value, Real)
        ):

            raise ValueError(
                f"Invalid probability for {family}."
            )

        probability = float(value)

        if (
            not math.isfinite(probability)
            or not 0 <= probability <= 1
        ):

            raise ValueError(
                f"Invalid probability for {family}."
            )

        probabilities[family] = probability

    if not math.isclose(
        sum(probabilities.values()),
        1.0,
        abs_tol=1e-6,
        rel_tol=0,
    ):

        raise ValueError(
            "Class probabilities must sum to one."
        )

    raw_attack_probability = sum(

        probabilities[family]

        for family in ATTACK_IMPACT
    )

    contributions = []

    if raw_attack_probability <= 1e-12:

        return {

            "expected_impact": 0.0,

            "raw_attack_probability":
                raw_attack_probability,

            "impact_contributions": [],

            "impact_available": False,

            "impact_policy_version":
                "attack_impact_v1",
        }

    expected_impact = 0.0

    for family, impact in ATTACK_IMPACT.items():

        conditional_probability = (

            probabilities[family]

            / raw_attack_probability
        )

        contribution = (

            conditional_probability * impact
        )

        expected_impact += contribution

        contributions.append({

            "attack_family": family,

            "class_probability":
                probabilities[family],

            "conditional_probability":
                conditional_probability,

            "impact_score": impact,

            "weighted_contribution":
                contribution,
        })

    contributions.sort(

        key=lambda item:
            item["weighted_contribution"],

        reverse=True,
    )

    return {

        "expected_impact":
            float(expected_impact),

        "raw_attack_probability":
            raw_attack_probability,

        "impact_contributions":
            contributions,

        "impact_available": True,

        "impact_policy_version":
            "attack_impact_v1",

        "class_probabilities_calibrated":
            False,
    }

def get_device_context(
    device_id: str | None = None,
) -> dict:

    if device_id is not None:

        if not isinstance(device_id, str):

            raise ValueError(
                "device_id must be a string."
            )

        if not device_id.strip():

            raise ValueError(
                "device_id cannot be empty."
            )

    if device_id in DEVICE_PROFILES:

        context = deepcopy(
            DEVICE_PROFILES[device_id]
        )

        context.update({

            "context_known": True,

            "context_source":
                "demonstration_inventory",

            "identity_verified": False,

            "simulation_only": True,
        })

        return context

    context = deepcopy(
        UNKNOWN_DEVICE_PROFILE
    )

    context.update({

        "requested_device_id": device_id,

        "context_known": False,

        "context_source":
            "unknown_device_fallback",

        "identity_verified": False,

        "simulation_only": True,
    })

    return context


# --------------------------------------------------
# LIST DEMONSTRATION DEVICES
# --------------------------------------------------

def list_device_profiles() -> list[dict]:

    return [

        get_device_context(device_id)

        for device_id in DEVICE_PROFILES
    ]

            status_code=422,

import {
  Injectable,
  inject,
  signal,
  computed,
} from '@angular/core';

import { finalize } from 'rxjs';

import { DeviceApiService } from './device-api.service';

import type {
  DeviceContextProfile,
} from '../models/device.model';

import {
  getApiErrorMessage,
} from '../../shared/utilities/api-error';

@Injectable({
  providedIn: 'root',
})
export class DeviceContextService {

  private readonly api = inject(DeviceApiService);

  private readonly profileState =
    signal<DeviceContextProfile[]>([]);

  private readonly selectedIdState =
    signal<string | null>(null);

  private readonly loadingState = signal(false);

  private readonly loadedState = signal(false);

  private readonly errorState = signal('');

  readonly profiles = this.profileState.asReadonly();

  readonly selectedDeviceId =
    this.selectedIdState.asReadonly();

  readonly loading = this.loadingState.asReadonly();

  readonly loaded = this.loadedState.asReadonly();

  readonly error = this.errorState.asReadonly();

  readonly selectedProfile = computed(() => {

    const selectedId = this.selectedIdState();

    return this.profileState().find(
      profile => profile.device_id === selectedId
    ) ?? null;

  });

  readonly totalProfiles = computed(
    () => this.profileState().length
  );

  readonly restrictedProfiles = computed(
    () => this.profileState().filter(
      profile => !profile.containment_allowed
    ).length
  );

  readonly verifiedProfiles = computed(
    () => this.profileState().filter(
      profile => profile.identity_verified
    ).length
  );

  loadProfiles(force = false): void {

    if (
      this.loadingState() ||
      (this.loadedState() && !force)
    ) {
      return;
    }

    this.loadingState.set(true);
    this.errorState.set('');

    this.api.getDeviceProfiles().pipe(

      finalize(() => this.loadingState.set(false))

    ).subscribe({

      next: response => {

        this.profileState.set(response.devices);
        this.loadedState.set(true);

        const selectedId = this.selectedIdState();

        if (
          selectedId !== null &&
          !response.devices.some(
            profile => profile.device_id === selectedId
          )
        ) {
          this.selectedIdState.set(null);
        }

      },

      error: error => {

        this.profileState.set([]);
        this.selectedIdState.set(null);
        this.loadedState.set(false);

        this.errorState.set(
          getApiErrorMessage(error)
        );

      },

    });

  }

  selectDevice(deviceId: string | null): boolean {

    if (deviceId === null) {

      this.selectedIdState.set(null);

      return true;

    }

    if (
      this.loadingState() ||
      !this.loadedState() ||
      !this.profileState().some(
        profile => profile.device_id === deviceId
      )
    ) {
      return false;
    }

    this.selectedIdState.set(deviceId);

    return true;

  }

  clearSelection(): void {

    this.selectedIdState.set(null);

  }

}

import {
  Component,
  DestroyRef,
  OnInit,
  inject,
  signal,
  computed,
} from '@angular/core';

import { FormsModule } from '@angular/forms';
import { DecimalPipe } from '@angular/common';

import {
  DeviceContextSelector,
} from '../../shared/components/device-context-selector/device-context-selector';
import { takeUntilDestroyed } from '@angular/core/rxjs-interop';
import { finalize } from 'rxjs';

import {
  MulticlassApiService,
} from '../../core/services/multiclass-api.service';

import {
  NotificationService,
} from '../../core/services/notification.service';

import {
  ATTACK_LABELS,
} from '../../core/models/prediction.model';

import type {
  ModelInfo,
  TrafficFeatures,
  AttackFamily,
  MulticlassPredictionResponse,
} from '../../core/models/prediction.model';

import {
  getApiErrorMessage,
} from '../../shared/utilities/api-error';

@Component({
  selector: 'app-detection',
  standalone: true,
  imports: [
    FormsModule,
    DecimalPipe,
    DeviceContextSelector,
  ],
  templateUrl: './detection.html',
  styleUrl: './detection.scss',
})
export class Detection implements OnInit {

  private readonly api = inject(MulticlassApiService);

  private readonly notifications =
    inject(NotificationService);

  private readonly destroyRef = inject(DestroyRef);

  protected readonly modelInfo =
    signal<ModelInfo | null>(null);

  protected readonly modelLoading = signal(false);

  protected readonly modelError = signal('');

  protected readonly submitting = signal(false);

  protected readonly errorMessage = signal('');

  protected readonly result =
    signal<MulticlassPredictionResponse | null>(null);

  protected readonly families =
    Object.values(ATTACK_LABELS);

  protected trafficJson = '';

  protected readonly statusLabels: Record<string, string> = {
    low_probability: 'Low attack probability',
    suspicious: 'Suspicious traffic',
    elevated_suspicion: 'Elevated suspicion',
    high_confidence_attack: 'High-confidence attack',
  };

  protected readonly probabilityRows = computed(() => {

    const prediction = this.result();

    if (!prediction) {
      return [];
    }

    return this.families.map(family => ({
      family,
      probability: prediction.class_probabilities[family],
      predicted: prediction.predicted_family === family,
    }));

  });

  protected readonly unsupportedClass = computed(() => {

    const prediction = this.result();
    const model = this.modelInfo();

    if (!prediction || !model) {
      return false;
    }

    return !model.validation_supported_labels.includes(
      prediction.predicted_label
    );

  });

  ngOnInit(): void {
    this.loadModelInfo();
  }

  protected loadModelInfo(): void {

    if (this.modelLoading() || this.submitting()) {
      return;
    }

    this.modelLoading.set(true);
    this.modelError.set('');
    this.modelInfo.set(null);

    this.api.getModelInfo().pipe(
      takeUntilDestroyed(this.destroyRef),
      finalize(() => this.modelLoading.set(false))
    ).subscribe({

      next: info => {

        if (
          !Array.isArray(info.features) ||
          info.feature_count !== 36 ||
          info.features.length !== 36 ||
          info.class_count !== 8
        ) {
          this.modelError.set(
            'Unexpected model configuration.'
          );
          return;
        }

        this.modelInfo.set(info);

      },

      error: error => {
        this.modelError.set(getApiErrorMessage(error));
      },

    });

  }

  protected loadTemplate(): void {

    const model = this.modelInfo();

    if (!model || this.submitting()) {
      return;
    }

    const template = Object.fromEntries(
      model.features.map(feature => [feature, null])
    );

    this.trafficJson = JSON.stringify(
      template, null, 2
    );

    this.result.set(null);
    this.errorMessage.set('');

    this.notifications.info(
      'Replace every null value with a measured numerical feature.',
      'Feature template loaded'
    );

  }

  protected clearInput(): void {

    if (this.submitting()) {
      return;
    }

    this.trafficJson = '';
    this.errorMessage.set('');
    this.result.set(null);

  }

  protected onInputChange(value: string): void {

    this.trafficJson = value;
    this.errorMessage.set('');
    this.result.set(null);

  }

  private validateFeatures(): TrafficFeatures {

    const model = this.modelInfo();

    if (!model) {
      throw new Error(
        'Model information is unavailable.'
      );
    }

    let parsed: unknown;

    try {
      parsed = JSON.parse(this.trafficJson);
    } catch {
      throw new Error(
        'Invalid JSON. Enter a valid traffic feature object.'
      );
    }

    if (
      typeof parsed !== 'object' ||
      parsed === null ||
      Array.isArray(parsed)
    ) {
      throw new Error(
        'Traffic features must be a JSON object.'
      );
    }

    const input = parsed as Record<string, unknown>;

    const missing = model.features.filter(
      feature => !Object.prototype.hasOwnProperty.call(
        input, feature
      )
    );

    if (missing.length > 0) {
      throw new Error(
        `Missing features: ${missing.join(', ')}`
      );
    }

    const extra = Object.keys(input).filter(
      feature => !model.features.includes(feature)
    );

    if (extra.length > 0) {
      throw new Error(
        `Unexpected features: ${extra.join(', ')}`
      );
    }

    const invalid = model.features.filter(feature => {

      const value = input[feature];

      return (
        typeof value !== 'number' ||
        !Number.isFinite(value) ||
        Math.abs(value) > 3.4028234663852886e38
      );

    });

    if (invalid.length > 0) {
      throw new Error(
        `Invalid numerical values: ${invalid.join(', ')}`
      );
    }

    const ordered = Object.fromEntries(
      model.features.map(feature => [
        feature,
        input[feature],
      ])
    );

    return ordered as unknown as TrafficFeatures;

  }

  private validateResponse(
    response: MulticlassPredictionResponse
  ): boolean {

    const model = this.modelInfo();

    const validProbability = (value: unknown): boolean =>
      typeof value === 'number' &&
      Number.isFinite(value) &&
      value >= 0 &&
      value <= 1;

    if (
      !response ||
      !model ||
      response.model_version !== model.model_version ||
      ATTACK_LABELS[response.predicted_label] !==
        response.predicted_family ||
      !validProbability(response.prediction_confidence) ||
      !validProbability(response.raw_attack_probability) ||
      !validProbability(response.calibrated_attack_probability) ||
      !response.class_probabilities ||
      !this.families.every(family =>
        validProbability(
          response.class_probabilities[family]
        )
      ) ||
      !Object.prototype.hasOwnProperty.call(
        this.statusLabels,
        response.detection_status
      ) ||
      response.mode !== 'prediction_only' ||
      response.incident_created !== false ||
      response.response_executed !== false
    ) {
      return false;
    }

    const sum = this.families.reduce(
      (total, family) =>
        total + response.class_probabilities[family],
      0
    );

    return Math.abs(sum - 1) < 0.0001;

  }

  protected detectTraffic(): void {

    if (this.submitting() || !this.modelInfo()) {
      return;
    }

    this.errorMessage.set('');
    this.result.set(null);

    let features: TrafficFeatures;

    try {
      features = this.validateFeatures();
    } catch (error) {

      this.errorMessage.set(
        error instanceof Error
          ? error.message
          : 'Invalid traffic features.'
      );

      return;

    }

    this.submitting.set(true);

    this.api.predict({
      features,
      source: 'manual',
    }).pipe(

      takeUntilDestroyed(this.destroyRef),

      finalize(() => this.submitting.set(false))

    ).subscribe({

      next: response => {

        if (!this.validateResponse(response)) {

          this.errorMessage.set(
            'Unexpected prediction response. Check the backend model and API contract.'
          );

          return;

        }

        this.result.set(response);

        if (response.high_confidence_alert) {

          this.notifications.warning(
            `Predicted family: ${response.predicted_family}. Further assessment is recommended.`,
            'High-confidence attack'
          );

        } else {

          this.notifications.success(
            'Multiclass prediction completed.',
            'Analysis completed'
          );

        }

      },

      error: error => {

        this.errorMessage.set(
          getApiErrorMessage(error)
        );

      },

    });

  }

}
